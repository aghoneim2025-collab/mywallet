// This is your Prisma schema file,
// learn more about it in the docs: https://pris.ly/d/prisma-schema

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

// ===== User (المستخدم - تسجيل دخول) =====
model User {
  id            String   @id @default(cuid())
  email         String   @unique
  name          String?
  passwordHash  String
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  accounts      Account[]
  categories    Category[]
  transactions  Transaction[]
  budgets       Budget[]

  @@map("users")
}

// ===== Account (الحسابات) =====
model Account {
  id        String   @id @default(cuid())
  name      String
  type      AccountType
  currency  Currency
  balance   Float    @default(0)
  color     String   @default("#2563EB")
  icon      String   @default("wallet")
  isDefault Boolean  @default(false)
  userId    String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  user          User?         @relation(fields: [userId], references: [id], onDelete: Cascade)
  transactions  Transaction[]
  transfersTo   Transaction[] @relation("TransfersTo")

  @@map("accounts")
}

enum AccountType {
  CASH         // نقدي
  BANK         // بنك
  E_WALLET     // محفظة إلكترونية
  CARD         // بطاقة ائتمان
  SAVINGS      // حساب توفير
  INVESTMENT   // استثمار (ذهب، كريبتو، أسهم، صناديق)
  LOAN         // قرض / سلفة
  EXTERNAL     // حساب خارجي (متابعة فقط - زي عقار، حساب شخص تاني)
}

enum Currency {
  EGP
  SAR
  USD
  EUR
  AED
}

// ===== Category (الفئات: مصروفات + دخل) =====
model Category {
  id        String        @id @default(cuid())
  nameAr    String
  nameEn    String
  type      CategoryType
  icon      String        @default("tag")
  color     String        @default("#10B981")
  isDefault Boolean       @default(false)
  userId    String?
  createdAt DateTime      @default(now())
  updatedAt DateTime      @updatedAt

  user         User?         @relation(fields: [userId], references: [id], onDelete: Cascade)
  transactions Transaction[]
  budgets     Budget[]

  @@map("categories")
}

enum CategoryType {
  EXPENSE
  INCOME
  TRANSFER   // تحويل بين حسابين (لا يحسب كدخل أو مصروف)
}

// ===== Transaction (المعاملة: مصروف أو دخل أو تحويل) =====
model Transaction {
  id            String   @id @default(cuid())
  type          CategoryType
  amount        Float
  currency      Currency
  accountId     String
  categoryId    String
  toAccountId   String?
  date          DateTime  @default(now())
  note          String?
  tags          String?
  transferPairId String?
  userId        String?
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  user         User?     @relation(fields: [userId], references: [id], onDelete: Cascade)
  account      Account   @relation(fields: [accountId], references: [id], onDelete: Cascade)
  category     Category  @relation(fields: [categoryId], references: [id], onDelete: Cascade)
  toAccount    Account?  @relation("TransfersTo", fields: [toAccountId], references: [id], onDelete: Cascade, onUpdate: Cascade)

  @@index([accountId])
  @@index([toAccountId])
  @@index([categoryId])
  @@index([date])
  @@index([type])
  @@index([userId])
  @@map("transactions")
}

// ===== Budget (الميزانية الشهرية لكل فئة) =====
model Budget {
  id          String   @id @default(cuid())
  categoryId  String
  month       String
  limit       Float
  currency    Currency   @default(EGP)
  userId      String?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  user        User?     @relation(fields: [userId], references: [id], onDelete: Cascade)
  category    Category  @relation(fields: [categoryId], references: [id], onDelete: Cascade)

  @@unique([categoryId, month])
  @@index([month])
  @@index([userId])
  @@map("budgets")
}

// ===== Setting (الإعدادات العامة) =====
model Setting {
  id        String   @id @default(cuid())
  key       String   @unique
  value     String
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("settings")
}

// ===== RecurringTransaction (معاملات متكررة - #1) =====
model RecurringTransaction {
  id          String   @id @default(cuid())
  type        String   // EXPENSE | INCOME | TRANSFER
  amount      Float
  currency    String
  accountId   String
  categoryId  String
  toAccountId String?  // للتحويلات
  frequency   String   // daily | weekly | monthly | yearly
  dayOfMonth  Int?     // يوم التنفيذ (1-31)
  startDate   DateTime
  endDate     DateTime?
  lastRun     DateTime?
  nextRun     DateTime
  note        String?
  active      Boolean  @default(true)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@map("recurring_transactions")
}

// ===== SavingsGoal (أهداف ادخارية - #7) =====
model SavingsGoal {
  id          String   @id @default(cuid())
  name        String
  targetAmount Float
  currentAmount Float @default(0)
  currency    String   @default("EGP")
  targetDate  DateTime?
  color       String   @default("#10B981")
  icon        String   @default("piggy-bank")
  achieved    Boolean  @default(false)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@map("savings_goals")
}

// ===== BillReminder (تذكير فواتير - #8) =====
model BillReminder {

  dueDate     DateTime

  accountId   String?
  paid        Boolean  @default(false)
  remindDays  Int      @default(3) // تذكير قبل كام يوم
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@map("bill_reminders")
}

'use client';
import { useState, useEffect } from 'react';
import { BottomNav, type TabKey } from '@/components/bottom-nav';
import { FAB } from '@/components/fab';
import { HomeScreen } from '@/components/home-screen';
import { CategoriesScreen } from '@/components/categories-screen';
import { AccountsScreen } from '@/components/accounts-screen';
import { ReportsScreen } from '@/components/reports-screen';
import { SettingsScreen } from '@/components/settings-screen';
import { BudgetScreen } from '@/components/budget-screen';
import { TransactionModal } from '@/components/transaction-modal';
import { TransferModal } from '@/components/transfer-modal';
import { AuthScreen } from '@/components/auth-screen';

export default function Home() {
  const [authed, setAuthed] = useState(false);
  const [checking, setChecking] = useState(true);
  const [activeTab, setActiveTab] = useState<TabKey>('home');
  const [txModal, setTxModal] = useState<{ open: boolean; type: 'EXPENSE' | 'INCOME'; initial?: any }>({ open: false, type: 'EXPENSE' });
  const [transferModal, setTransferModal] = useState(false);

  // Check auth on mount
  useEffect(() => {
    fetch('/api/auth/session')
      .then(r => r.json())
      .then(d => {
        if (d.data) setAuthed(true);
      })
      .catch(() => {})
      .finally(() => setChecking(false));
  }, []);

  const openAddExpense = () => setTxModal({ open: true, type: 'EXPENSE' });
  const openAddIncome = () => setTxModal({ open: true, type: 'INCOME' });
  const openEditTx = (tx: any) => setTxModal({ open: true, type: tx.type === 'INCOME' ? 'INCOME' : 'EXPENSE', initial: tx });
  const openTransfer = () => setTransferModal(true);

  const [reloadKey, setReloadKey] = useState(0);
  const handleSaved = () => setReloadKey(k => k + 1);

  // Show loading while checking auth
  if (checking) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-blue-600">
        <div className="w-12 h-12 border-4 border-white/30 border-t-white rounded-full animate-spin" />
      </div>
    );
  }

  // Show auth screen if not logged in
  if (!authed) {
    return <AuthScreen onAuth={() => { setAuthed(true); setReloadKey(k => k + 1); }} />;
  }

  return (
    <div className="flex flex-col min-h-screen">
      <main className="flex-1 pb-24">
        {activeTab === 'home' && (
          <HomeScreen key={`h-${reloadKey}`} onEditTx={openEditTx} onOpenAccounts={() => setActiveTab('accounts')} />
        )}
        {activeTab === 'categories' && <CategoriesScreen key={`c-${reloadKey}`} />}
        {activeTab === 'accounts' && <AccountsScreen key={`a-${reloadKey}`} />}
        {activeTab === 'reports' && <ReportsScreen key={`r-${reloadKey}`} />}
        {activeTab === 'settings' && (
          <>
            <SettingsScreen onLogout={() => { setAuthed(false); setActiveTab('home'); }} />
            <div className="px-4 pb-4">
              <BudgetScreen key={`b-${reloadKey}`} />
            </div>
          </>
        )}
      </main>

      <FAB onAddExpense={openAddExpense} onAddIncome={openAddIncome} onTransfer={openTransfer} />
      <BottomNav active={activeTab} onChange={setActiveTab} />

      <TransactionModal
        open={txModal.open}
        onOpenChange={(o) => setTxModal(m => ({ ...m, open: o }))}
        initial={txModal.initial || { type: txModal.type }}
        onSaved={handleSaved}
      />
      <TransferModal open={transferModal} onOpenChange={setTransferModal} onSaved={handleSaved} />
    </div>
  );
}

// نظام العملات المتعدد مع تحويل
// أسعار الصرف قابلة للتحديث عبر الإعدادات (#9)

export type Currency = 'EGP' | 'SAR' | 'USD' | 'EUR' | 'AED';

// سعر الصرف مقابل EGP (1 وحدة من العملة = كم EGP)
// القيم الافتراضية - يمكن تحديثها من الإعدادات
export const DEFAULT_EXCHANGE_RATES: Record<Currency, number> = {
  EGP: 1,
  SAR: 13.0,
  USD: 48.5,
  EUR: 52.0,
  AED: 13.2,
};

// الأسعار الحالية - تتحدث من localStorage أو الإعدادات
export let EXCHANGE_RATES: Record<Currency, number> = { ...DEFAULT_EXCHANGE_RATES };

// تحديث الأسعار من الإعدادات
export function updateExchangeRates(rates: Partial<Record<Currency, number>>) {
  EXCHANGE_RATES = { ...EXCHANGE_RATES, ...rates };
}

// تحميل الأسعار من localStorage (client-side)
export function loadExchangeRates(): Record<Currency, number> {
  if (typeof window !== 'undefined') {
    try {
      const saved = localStorage.getItem('mw-exchange-rates');
      if (saved) {
        const parsed = JSON.parse(saved);
        EXCHANGE_RATES = { ...DEFAULT_EXCHANGE_RATES, ...parsed };
      }
    } catch {}
  }
  return EXCHANGE_RATES;
}

// حفظ الأسعار في localStorage
export function saveExchangeRates(rates: Record<Currency, number>) {
  EXCHANGE_RATES = rates;
  if (typeof window !== 'undefined') {
    localStorage.setItem('mw-exchange-rates', JSON.stringify(rates));
  }
}

export const CURRENCY_SYMBOLS: Record<Currency, { ar: string; en: string }> = {
  EGP: { ar: 'ج.م', en: 'E£' },
  SAR: { ar: '﷼', en: 'SAR' },
  USD: { ar: '$', en: '$' },
  EUR: { ar: '€', en: '€' },
  AED: { ar: 'د.إ', en: 'AED' },
};

// تحويل مبلغ من عملة لأخرى عبر EGP كعملة وسيطة
export function convert(amount: number, from: Currency, to: Currency): number {
  if (from === to) return amount;
  const inEgp = amount * EXCHANGE_RATES[from];
  return inEgp / EXCHANGE_RATES[to];
}

// تنسيق المبلغ مع رمز العملة
export function formatMoney(
  amount: number,
  currency: Currency,
  lang: 'ar' | 'en' = 'ar'
): string {
  const sym = CURRENCY_SYMBOLS[currency][lang];
  const formatted = new Intl.NumberFormat(lang === 'ar' ? 'ar-EG' : 'en-US', {
    minimumFractionDigits: 0,
    maximumFractionDigits: 2,
  }).format(Math.abs(amount));
  const sign = amount < 0 ? '-' : '';
  return lang === 'ar'
    ? `${sign}${formatted} ${sym}`
    : `${sign}${sym} ${formatted}`;
}

// اختصار رقم كبير (مثلا 15000 -> 15K)
export function compactNumber(amount: number, lang: 'ar' | 'en' = 'ar'): string {
  return new Intl.NumberFormat(lang === 'ar' ? 'ar-EG' : 'en-US', {
    notation: 'compact',
    maximumFractionDigits: 1,
  }).format(amount);
}

// Simple hash function (no bcrypt needed for SQLite/standalone)
// Uses Node.js crypto for SHA-256 + salt
import { createHash, randomBytes } from 'crypto';

export function hashPassword(password: string): string {
  const salt = randomBytes(16).toString('hex');
  const hash = createHash('sha256').update(salt + password).digest('hex');
  return `${salt}:${hash}`;
}

export function verifyPassword(password: string, stored: string): boolean {
  const [salt, hash] = stored.split(':');
  const testHash = createHash('sha256').update(salt + password).digest('hex');
  return hash === testHash;
}

export function generateToken(userId: string): string {
  const payload = JSON.stringify({ userId, exp: Date.now() + 30 * 24 * 60 * 60 * 1000 });
  return Buffer.from(payload).toString('base64url');
}

export function verifyToken(token: string): { userId: string } | null {
  try {
    const payload = JSON.parse(Buffer.from(token, 'base64url').toString());
    if (payload.exp < Date.now()) return null;
    return { userId: payload.userId };
  } catch {
    return null;
  }
}

/**
 * Get user ID from request (via Authorization header or cookie)
 * Returns null if not authenticated
 */
export function getUserIdFromRequest(req: Request): string | null {
  // Try Authorization header
  const authHeader = req.headers.get('authorization');
  if (authHeader?.startsWith('Bearer ')) {
    const token = authHeader.slice(7);
    const result = verifyToken(token);
    if (result) return result.userId;
  }
  // Try cookie
  const cookie = req.headers.get('cookie') || '';
  const tokenMatch = cookie.match(/auth-token=([^;]+)/);
  if (tokenMatch) {
    const result = verifyToken(tokenMatch[1]);
    if (result) return result.userId;
  }
  return null;
}

// Simple fetch wrapper
const baseUrl = '';

export async function api<T>(
  path: string,
  opts?: RequestInit & { params?: Record<string, string | undefined> }
): Promise<T> {
  const { params, ...rest } = opts || {};
  let url = path;
  if (params) {
    const sp = new URLSearchParams();
    Object.entries(params).forEach(([k, v]) => {
      if (v !== undefined) sp.set(k, v);
    });
    const qs = sp.toString();
    if (qs) url += `?${qs}`;
  }

  const res = await fetch(`${baseUrl}${url}`, {
    headers: { 'Content-Type': 'application/json', ...(rest?.headers || {}) },
    ...rest,
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: 'Network error' }));
    throw new Error(err.error || res.statusText);
  }

  return res.json();
}

// Helper: download a file (for CSV export)
export async function downloadFile(url: string, filename: string) {
  const res = await fetch(url);
  const blob = await res.blob();
  const blobUrl = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = blobUrl;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(blobUrl);
}

      const today = new Date().toISOString();

      setItems(r.data);

  // Date range filter
  const dateRange = useMemo(() => {
    const now = new Date();
    const end = now.toISOString().slice(0, 10);
    let start: string;
    switch (range) {
      case 'today': start = end; break;
      case 'week':
        const w = new Date(now); w.setDate(w.getDate() - 7);
        start = w.toISOString().slice(0, 10); break;
      case 'month':
        start = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`; break;

      const txDate = new Date(tx.date).toISOString().slice(0, 10);
      if (txDate < dateRange.start || txDate > dateRange.end) return;

        const q = search.toLowerCase();
        const name = (lang === 'ar' ? tx.category.nameAr : tx.category.nameEn).toLowerCase();
        if (!name.includes(q)) return;
      }
      const cid = tx.category.id;
      if (!map.has(cid)) map.set(cid, { category: tx.category, total: 0, count: 0 });
      const entry = map.get(cid)!;
      entry.total += convert(tx.amount, tx.currency as Currency, baseCurrency);
      entry.count += 1;
    });
    return Array.from(map.values()).sort((a, b) => b.total - a.total);

      setAccounts(r.data);
    } catch (e: any) {
      toast.error(e.message);
    } finally { setLoading(false); }

    convert(balance, fromCurrency as Currency, baseCurrency);

  // Filter accounts by search
  const filteredAccounts = accounts.filter(a => {
    if (!search) return true;
    const q = search.toLowerCase();
    return a.name.toLowerCase().includes(q) ||
      a.currency.toLowerCase().includes(q) ||
      a.type.toLowerCase().includes(q);
  });

  const [stat, setStat] = useState<Stat | null>(null);
  const [loading, setLoading] = useState(true);
  const [month, setMonth] = useState(new Date().toISOString().slice(0, 7));
  const [viewMode, setViewMode] = useState<'monthly' | 'yearly'>('monthly');
  const [year, setYear] = useState(new Date().getFullYear().toString());
  const [yearlyData, setYearlyData] = useState<{ month: string; income: number; expense: number; balance: number }[] | null>(null);
  const [detailCategoryId, setDetailCategoryId] = useState<string | null>(null);
  const [lastMonthStat, setLastMonthStat] = useState<Stat | null>(null);

  const load = async () => {
    setLoading(true);
    try {
      if (viewMode === 'monthly') {
        const [s, lastM] = await Promise.all([
          api<{ data: Stat }>(`/api/stats?month=${month}&baseCurrency=${baseCurrency}`),
          // Fetch last month for comparison
          (async () => {
            const lastMonthDate = new Date(month + '-01');
            lastMonthDate.setMonth(lastMonthDate.getMonth() - 1);
            const lastMonthStr = lastMonthDate.toISOString().slice(0, 7);
            try {
              const r = await api<{ data: Stat }>(`/api/stats?month=${lastMonthStr}&baseCurrency=${baseCurrency}`);
              return r.data;
            } catch { return null; }
          })(),
        ]);
        setStat(s.data);
        setLastMonthStat(lastM);
      } else {
        // Yearly: fetch 12 months stats in parallel
        const months: { month: string; income: number; expense: number; balance: number }[] = [];
        const promises = [];
        for (let m = 1; m <= 12; m++) {
          const mStr = `${year}-${String(m).padStart(2, '0')}`;
          promises.push(
            api<{ data: Stat }>(`/api/stats?month=${mStr}&baseCurrency=${baseCurrency}`)
              .then(s => ({
                month: mStr,
                income: s.data.totalIncome,
                expense: s.data.totalExpense,
                balance: s.data.balance,
              }))
              .catch(() => ({ month: mStr, income: 0, expense: 0, balance: 0 }))
          );
        }
        const results = await Promise.all(promises);
        setYearlyData(results);
        // Also set stat to yearly totals
        const totalIncome = results.reduce((s, r) => s + r.income, 0);
        const totalExpense = results.reduce((s, r) => s + r.expense, 0);
        setStat({
          totalIncome,
          totalExpense,
          balance: totalIncome - totalExpense,
          totalAssets: 0,
          baseCurrency,
          expenseCount: 0,
          incomeCount: 0,
          byCategory: [],
          byDay: [],
          month: year,
        } as Stat);
      }
    } finally { setLoading(false); }
  };

  useEffect(() => {
    if (viewMode === 'monthly') {
      load();
    } else {
      load();
    }
  }, [month, baseCurrency, viewMode, year]);

  const fmt = (n: number) => formatMoney(n, baseCurrency, lang);

  const handleExportCSV = async () => {
    try {
      await downloadFile(`/api/excel?action=export&month=${month}`, `mywallet-${month}.xlsx`);
      toast.success(lang === 'ar' ? 'تم تصدير Excel' : 'Excel exported');
    } catch (e: any) {
      toast.error(e.message);
    }
  };

  // #6: PDF Export
  const handleExportPDF = async () => {
    try {
      const { jsPDF } = await import('jspdf');
      const autoTable = (await import('jspdf-autotable')).default;
      const doc = new jsPDF();

      // Title
      doc.setFontSize(20);
      doc.text(`MyWallet Report - ${month}`, 14, 20);
      doc.setFontSize(12);
      doc.text(`Base Currency: ${baseCurrency}`, 14, 30);

      // Summary
      doc.text(`Income:    ${fmt(stat.totalIncome)}`, 14, 45);
      doc.text(`Expense:   ${fmt(stat.totalExpense)}`, 14, 52);
      doc.text(`Balance:   ${fmt(stat.balance)}`, 14, 59);

      // Categories table
      autoTable(doc, {
        startY: 70,
        head: [['Category', 'Count', 'Amount', '%']],
        body: sortedCategories.map(c => [
          lang === 'ar' ? c.category.nameAr : c.category.nameEn,
          String(c.count),
          fmt(c.total),
          ((c.total / stat.totalExpense) * 100).toFixed(1) + '%',
        ]),
        theme: 'striped',
        headStyles: { fillColor: [37, 99, 235] },
      });

      doc.save(`mywallet-report-${month}.pdf`);
      toast.success(lang === 'ar' ? 'تم تصدير PDF' : 'PDF exported');
    } catch (e: any) {
      toast.error(e.message);
    }
  };

  if (loading || !stat) {
    return (
      <div className="p-4 space-y-3">
        <Skeleton className="h-12 w-full" />
        <Skeleton className="h-28 w-full" />
        <Skeleton className="h-64 w-full" />
      </div>
    );
  }

  // Pie chart data (top 8 categories + "Other")
  const sortedCategories = [...stat.byCategory].sort((a, b) => b.total - a.total);
  const topCategories = sortedCategories.slice(0, 8);
  const otherTotal = sortedCategories.slice(8).reduce((s, c) => s + c.total, 0);
  const pieData = [
    ...topCategories.map(c => ({
      name: lang === 'ar' ? c.category.nameAr : c.category.nameEn,
      value: c.total,
      color: c.category.color,
    })),
    ...(otherTotal > 0 ? [{
      name: lang === 'ar' ? 'أخرى' : 'Other',
      value: otherTotal,
      color: '#64748B',
    }] : []),
  ];

  // Daily bar chart
  const barData = stat.byDay.map(d => ({
    day: parseInt(d.date.split('-')[2], 10),
    expense: d.expense,
    income: d.income,
  }));

  // Savings rate
  const savingsRate = stat.totalIncome > 0
    ? ((stat.totalIncome - stat.totalExpense) / stat.totalIncome) * 100
    : 0;

  return (
    <div className="p-4 space-y-5">
      {/* Header */}
      <header className="flex justify-between items-center gap-2">
        <div>
          <h1 className="text-2xl font-bold">{t('reports')}</h1>
          <p className="text-xs text-slate-500">{viewMode === 'monthly' ? month : year}</p>
        </div>
        <div className="flex gap-1">
          {/* Toggle monthly/yearly */}
          <div className="flex gap-1 bg-slate-100 dark:bg-slate-800 rounded-lg p-1">
            <button
              onClick={() => setViewMode('monthly')}
              className={cn(
                'px-2 py-1 rounded text-xs font-medium transition',
                viewMode === 'monthly'
                  ? 'bg-white dark:bg-slate-900 text-blue-700 dark:text-blue-300 shadow-sm'
                  : 'text-slate-500'
              )}
            >
              {lang === 'ar' ? 'شهري' : 'Monthly'}
            </button>
            <button
              onClick={() => setViewMode('yearly')}
              className={cn(
                'px-2 py-1 rounded text-xs font-medium transition',
                viewMode === 'yearly'
                  ? 'bg-white dark:bg-slate-900 text-blue-700 dark:text-blue-300 shadow-sm'
                  : 'text-slate-500'
              )}
            >
              {lang === 'ar' ? 'سنوي' : 'Yearly'}
            </button>
          </div>
        </div>
      </header>

      {/* Date selector (month or year) */}
      <div>
        {viewMode === 'monthly' ? (
          <input
            type="month"
            value={month}
            onChange={e => setMonth(e.target.value)}
            className="bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg px-2 py-1.5 text-sm w-full"
          />
        ) : (
          <select
            value={year}
            onChange={e => setYear(e.target.value)}
            className="bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg px-2 py-1.5 text-sm w-full"
          >
            {Array.from({ length: 15 }, (_, i) => (new Date().getFullYear() - 7 + i).toString()).map(y => (
              <option key={y} value={y}>{y}</option>
            ))}
          </select>
        )}
      </div>

      {/* Summary cards - 3 columns */}
      <div className="grid grid-cols-3 gap-2">
        <Card className="p-3 bg-emerald-50 dark:bg-emerald-950/30 border-emerald-200 dark:border-emerald-900">
          <div className="flex items-center gap-1.5 mb-1">
            <TrendingUp className="w-3.5 h-3.5 text-emerald-600" />
            <span className="text-[10px] font-semibold text-emerald-700 dark:text-emerald-400">
              {t('monthlyIncome')}
            </span>
          </div>
          <p className="text-sm font-bold text-emerald-700 dark:text-emerald-300 truncate">{fmt(stat.totalIncome)}</p>
          <p className="text-[10px] text-emerald-600/70">{stat.incomeCount} {lang === 'ar' ? 'معاملة' : 'txns'}</p>
        </Card>
        <Card className="p-3 bg-red-50 dark:bg-red-950/30 border-red-200 dark:border-red-900">
          <div className="flex items-center gap-1.5 mb-1">
            <TrendingDown className="w-3.5 h-3.5 text-red-600" />
            <span className="text-[10px] font-semibold text-red-700 dark:text-red-400">
              {t('monthlyExpenses')}
            </span>
          </div>
          <p className="text-sm font-bold text-red-700 dark:text-red-300 truncate">{fmt(stat.totalExpense)}</p>
          <p className="text-[10px] text-red-600/70">{stat.expenseCount} {lang === 'ar' ? 'معاملة' : 'txns'}</p>
        </Card>
        <Card className="p-3 bg-blue-50 dark:bg-blue-950/30 border-blue-200 dark:border-blue-900">
          <div className="flex items-center gap-1.5 mb-1">
            <Wallet className="w-3.5 h-3.5 text-blue-600" />
            <span className="text-[10px] font-semibold text-blue-700 dark:text-blue-400">
              {lang === 'ar' ? 'الصافي' : 'Net'}
            </span>
          </div>
          <p className={cn(
            "text-sm font-bold truncate",
            stat.balance >= 0 ? "text-blue-700 dark:text-blue-300" : "text-red-700 dark:text-red-300"
          )}>
            {fmt(stat.balance)}
          </p>
          <p className="text-[10px] text-blue-600/70">
            {savingsRate >= 0 ? '+' : ''}{savingsRate.toFixed(1)}% {lang === 'ar' ? 'ادخار' : 'savings'}
          </p>
        </Card>
      </div>

      {/* Yearly chart - shows when viewMode = yearly */}
      {viewMode === 'yearly' && yearlyData && (
        <section>
          <h2 className="text-sm font-semibold mb-2">
            {lang === 'ar' ? 'الدخل مقابل المصروفات شهرياً' : 'Monthly Income vs Expenses'}
          </h2>
          <Card className="p-3">
            <ResponsiveContainer width="100%" height={220}>
              <BarChart
                data={yearlyData.map(d => ({
                  month: d.month.split('-')[1],
                  income: d.income,
                  expense: d.expense,
                }))}
              >
                <XAxis dataKey="month" tick={{ fontSize: 9, fill: '#94a3b8' }} axisLine={false} tickLine={false} />
                <YAxis tick={{ fontSize: 9, fill: '#94a3b8' }} axisLine={false} tickLine={false} width={40} />
                <Tooltip
                  formatter={(v: number) => fmt(v)}
                  contentStyle={{ fontSize: 11, borderRadius: 8, border: '1px solid #e2e8f0' }}
                />
                <Bar dataKey="income" name={t('income')} fill="#10B981" radius={[4, 4, 0, 0]} maxBarSize={20} />
                <Bar dataKey="expense" name={t('expense')} fill="#EF4444" radius={[4, 4, 0, 0]} maxBarSize={20} />
              </BarChart>
            </ResponsiveContainer>
          </Card>

          {/* Monthly breakdown table */}
          <Card className="p-3 mt-2">
            <div className="grid grid-cols-4 gap-2 text-[10px] font-semibold text-slate-500 mb-2">
              <span>{lang === 'ar' ? 'الشهر' : 'Month'}</span>
              <span className="text-end text-emerald-600">{lang === 'ar' ? 'دخل' : 'Income'}</span>
              <span className="text-end text-red-600">{lang === 'ar' ? 'مصروف' : 'Expense'}</span>
              <span className="text-end text-blue-600">{lang === 'ar' ? 'صافي' : 'Net'}</span>
            </div>
            <div className="divide-y divide-slate-100 dark:divide-slate-800">
              {yearlyData.map(d => (
                <div key={d.month} className="grid grid-cols-4 gap-2 py-1.5 text-xs">
                  <span className="text-slate-700 dark:text-slate-300">{d.month}</span>
                  <span className="text-end text-emerald-600 dark:text-emerald-400">{fmt(d.income)}</span>
                  <span className="text-end text-red-600 dark:text-red-400">{fmt(d.expense)}</span>
                  <span className={cn('text-end font-semibold', d.balance >= 0 ? 'text-blue-600 dark:text-blue-400' : 'text-red-600 dark:text-red-400')}>{fmt(d.balance)}</span>
                </div>
              ))}
            </div>
          </Card>
        </section>
      )}

      {/* #5: Monthly comparison - this month vs last month */}
      {viewMode === 'monthly' && lastMonthStat && (
        <section>
          <h2 className="text-sm font-semibold mb-2">
            {lang === 'ar' ? 'مقارنة بالشهر الماضي' : 'vs Last Month'}
          </h2>
          <Card className="p-3">
            <div className="grid grid-cols-3 gap-2 text-center">
              {/* Income comparison */}
              <div>
                <p className="text-[10px] text-slate-500">{t('income')}</p>
                <p className="text-xs font-bold text-emerald-600">{fmt(stat.totalIncome)}</p>
                {(() => {
                  const diff = stat.totalIncome - lastMonthStat.totalIncome;
                  const pct = lastMonthStat.totalIncome > 0 ? (diff / lastMonthStat.totalIncome) * 100 : 0;
                  return (
                    <p className={cn('text-[10px] font-medium', diff >= 0 ? 'text-emerald-600' : 'text-red-600')}>
                      {diff >= 0 ? '↑' : '↓'} {Math.abs(pct).toFixed(0)}%
                    </p>
                  );
                })()}
              </div>
              {/* Expense comparison */}
              <div>
                <p className="text-[10px] text-slate-500">{t('expense')}</p>
                <p className="text-xs font-bold text-red-600">{fmt(stat.totalExpense)}</p>
                {(() => {
                  const diff = stat.totalExpense - lastMonthStat.totalExpense;
                  const pct = lastMonthStat.totalExpense > 0 ? (diff / lastMonthStat.totalExpense) * 100 : 0;
                  return (
                    <p className={cn('text-[10px] font-medium', diff <= 0 ? 'text-emerald-600' : 'text-red-600')}>
                      {diff <= 0 ? '↓' : '↑'} {Math.abs(pct).toFixed(0)}%
                    </p>
                  );
                })()}
              </div>
              {/* Balance comparison */}
              <div>
                <p className="text-[10px] text-slate-500">{lang === 'ar' ? 'الصافي' : 'Net'}</p>
                <p className={cn('text-xs font-bold', stat.balance >= 0 ? 'text-blue-600' : 'text-red-600')}>{fmt(stat.balance)}</p>
                {(() => {
                  const diff = stat.balance - lastMonthStat.balance;
                  const pct = lastMonthStat.balance !== 0 ? (diff / Math.abs(lastMonthStat.balance)) * 100 : 0;
                  return (
                    <p className={cn('text-[10px] font-medium', diff >= 0 ? 'text-emerald-600' : 'text-red-600')}>
                      {diff >= 0 ? '↑' : '↓'} {Math.abs(pct).toFixed(0)}%
                    </p>
                  );
                })()}
              </div>
            </div>
          </Card>
        </section>
      )}

      {/* Daily trend bar chart - only in monthly mode */}
      {viewMode === 'monthly' && (
      <section>
        <h2 className="text-sm font-semibold mb-2">{t('incomeVsExpense')}</h2>
        <Card className="p-3">
          <ResponsiveContainer width="100%" height={200}>
            <BarChart data={barData}>
              <XAxis dataKey="day" tick={{ fontSize: 9, fill: '#94a3b8' }} axisLine={false} tickLine={false} interval={4} />
              <YAxis tick={{ fontSize: 9, fill: '#94a3b8' }} axisLine={false} tickLine={false} width={35} />
              <Tooltip
                formatter={(v: number) => fmt(v)}
                contentStyle={{ fontSize: 11, borderRadius: 8, border: '1px solid #e2e8f0' }}
              />
              <Bar dataKey="income" name={t('income')} fill="#10B981" radius={[4, 4, 0, 0]} maxBarSize={20} />
              <Bar dataKey="expense" name={t('expense')} fill="#EF4444" radius={[4, 4, 0, 0]} maxBarSize={20} />
            </BarChart>
          </ResponsiveContainer>
        </Card>
      </section>
      )}

      {/* Pie chart with category breakdown */}
      <section>
        <h2 className="text-sm font-semibold mb-2">{lang === 'ar' ? 'توزيع المصروفات' : 'Expense Distribution'}</h2>
        {pieData.length === 0 ? (
          <Card className="p-8 text-center">
            <p className="text-sm text-slate-400">{t('noTransactionsYet')}</p>
          </Card>
        ) : (
          <Card className="p-3">
            <div className="flex items-center gap-3">
              {/* Pie chart on the left */}
              <div className="w-1/3 shrink-0">
                <ResponsiveContainer width="100%" height={140}>
                  <PieChart>
                    <Pie
                      data={pieData}
                      dataKey="value"
                      nameKey="name"
                      cx="50%"
                      cy="50%"
                      innerRadius={32}
                      outerRadius={60}
                      paddingAngle={2}
                    >
                      {pieData.map((entry, i) => (
                        <Cell key={i} fill={entry.color} />
                      ))}
                    </Pie>
                    <Tooltip
                      formatter={(v: number) => fmt(v)}
                      contentStyle={{ fontSize: 11, borderRadius: 8, border: '1px solid #e2e8f0' }}
                    />
                  </PieChart>
                </ResponsiveContainer>
              </div>
              {/* Center: Total in the middle of donut */}
              <div className="absolute left-0 top-1/2 -translate-y-1/2 w-1/3 text-center pointer-events-none">
                <p className="text-[10px] text-slate-400">{lang === 'ar' ? 'الإجمالي' : 'Total'}</p>
                <p className="text-sm font-bold">{fmt(stat.totalExpense)}</p>
              </div>
              {/* Legend on the right */}
              <div className="flex-1 space-y-1.5">
                {pieData.map((entry, i) => {
                  const pct = stat.totalExpense > 0 ? (entry.value / stat.totalExpense) * 100 : 0;
                  return (
                    <div key={i} className="flex items-center gap-2">
                      <div
                        className="w-2.5 h-2.5 rounded-full shrink-0"
                        style={{ background: entry.color }}
                      />
                      <span className="text-[11px] font-medium flex-1 truncate">{entry.name}</span>
                      <span className="text-[10px] text-slate-500">{pct.toFixed(1)}%</span>
                      <span className="text-[11px] font-semibold">{fmt(entry.value)}</span>
                    </div>
                  );
                })}
              </div>
            </div>
          </Card>
        )}
      </section>

      {/* Top categories list */}
      <section>
        <h2 className="text-sm font-semibold mb-2">{t('topCategories')}</h2>
        {sortedCategories.length === 0 ? (
          <Card className="p-6 text-center">
            <p className="text-sm text-slate-400">{t('noTransactionsYet')}</p>
          </Card>
        ) : (
          <div className="space-y-2">
            {sortedCategories.slice(0, 10).map((c, i) => {
              const Icon = getIcon(c.category.icon);
              const pct = stat.totalExpense > 0 ? (c.total / stat.totalExpense) * 100 : 0;
              return (
                <Card key={i} className="p-3 cursor-pointer hover:shadow-md transition" onClick={() => setDetailCategoryId(c.category.id)}>
                  <div className="flex items-center gap-3">
                    <div
                      className="w-9 h-9 rounded-full flex items-center justify-center text-white shrink-0"
                      style={{ background: c.category.color }}
                    >
                      <Icon className="w-4 h-4" />
                    </div>
                    <div className="flex-1 min-w-0">
                      <div className="flex justify-between items-baseline mb-1">
                        <span className="text-sm font-medium truncate">
                          {lang === 'ar' ? c.category.nameAr : c.category.nameEn}
                        </span>
                        <span className="text-sm font-semibold shrink-0">{fmt(c.total)}</span>
                      </div>
                      <div className="flex items-center gap-2">
                        <div className="h-1.5 bg-slate-100 dark:bg-slate-800 rounded-full overflow-hidden flex-1">
                          <div
                            className="h-full rounded-full transition-all"
                            style={{ width: `${pct}%`, background: c.category.color }}
                          />
                        </div>
                        <span className="text-[10px] text-slate-500 shrink-0">{pct.toFixed(1)}%</span>
                      </div>
                    </div>
                  </div>
                </Card>
              );
            })}
          </div>
        )}
      </section>

      {/* Export */}
      <section className="grid grid-cols-2 gap-2">
        <Button
          onClick={handleExportCSV}
          variant="outline"
          className="h-12 rounded-xl bg-emerald-50 dark:bg-emerald-950/30 border-emerald-200 dark:border-emerald-900 text-emerald-700 dark:text-emerald-300 hover:bg-emerald-100"
        >
          <Download className="w-5 h-5" /> Excel
        </Button>
        <Button
          onClick={handleExportPDF}
          variant="outline"
          className="h-12 rounded-xl bg-red-50 dark:bg-red-950/30 border-red-200 dark:border-red-900 text-red-700 dark:text-red-300 hover:bg-red-100"
        >
          <FileText className="w-5 h-5" /> PDF
        </Button>
      </section>

      {/* Category Detail Modal */}
      {detailCategoryId && (
        <CategoryDetailModal
          categoryId={detailCategoryId}
          month={month}
          onClose={() => setDetailCategoryId(null)}
        />
      )}
    </div>
  );
}

/** #4: Category Detail Modal - shows all transactions for a category in a given month */
function CategoryDetailModal({
  categoryId, month, onClose,
}: {
  categoryId: string;
  month: string;
  onClose: () => void;
}) {
  const { t, lang } = useT();
  const baseCurrency = useSettings(s => s.baseCurrency);
  const [txs, setTxs] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const start = `${month}-01`;
    const end = new Date(start);
    end.setMonth(end.getMonth() + 1);
    api<{ data: any[] }>(`/api/transactions?categoryId=${categoryId}&from=${start}&to=${end.toISOString()}&limit=500`)
      .then(r => setTxs(r.data))
      .finally(() => setLoading(false));
  }, [categoryId, month]);

  const fmt = (n: number, cur: any = baseCurrency) => formatMoney(n, cur, lang);
  const total = txs.reduce((s, t) => s + convert(t.amount, t.currency as Currency, baseCurrency), 0);

  return (
    <Dialog open={true} onOpenChange={(o) => !o && onClose()}>
      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>
            {txs[0] ? (lang === 'ar' ? txs[0].category.nameAr : txs[0].category.nameEn) : (lang === 'ar' ? 'تفاصيل الفئة' : 'Category Details')}
          </DialogTitle>
        </DialogHeader>
        {loading ? (
          <Skeleton className="h-40 w-full" />
        ) : (
          <>
            <Card className="p-4 bg-blue-600 text-white border-0">
              <p className="text-xs text-blue-100">{lang === 'ar' ? 'إجمالي الإنفاق' : 'Total Spending'}</p>
              <p className="text-2xl font-bold">{fmt(total)}</p>
              <p className="text-xs text-blue-200 mt-1">{txs.length} {lang === 'ar' ? 'معاملة' : 'transactions'}</p>
            </Card>
            <div className="mt-3 space-y-1.5 max-h-72 overflow-y-auto">
              {txs.map(tx => {
                const Icon = getIcon(tx.category?.icon || 'tag');
                const isIncome = tx.type === 'INCOME';

  open,
  onOpenChange,
  initial,
  onSaved,

  open: boolean;
  onOpenChange: (o: boolean) => void;
  initial?: Tx | null;
  onSaved?: () => void;

  const [date, setDate] = useState('');
  const [note, setNote] = useState('');
  const [categories, setCategories] = useState<Category[]>([]);
  const [accounts, setAccounts] = useState<Account[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

      // load categories + accounts

        api<{ data: Category[] }>('/api/categories'),
        api<{ data: Account[] }>('/api/accounts'),
      ]).then(([c, a]) => {
        setCategories(c.data);
        setAccounts(a.data);
        const def = a.data.find(x => (x as any).isDefault) || a.data[0];
        const chosenAccId = initial?.accountId || def?.id || a.data[0]?.id || '';
        setAccountId(chosenAccId);
        // لو معاملة موجودة، استخدم عملتها الأصلية. لو جديدة، استخدم عملة الحساب المختار
        if (initial?.currency) {
          setCurrency(initial.currency);
        } else {
          const chosenAcc = a.data.find(x => x.id === chosenAccId);
          setCurrency((chosenAcc?.currency as Currency) || baseCurrency);
        }
      });

interface Account {
  id: string;
  name: string;
  type: string;
  currency: string;
  balance: number;
  color: string;
  icon: string;
}

export function TransferModal({
  open,
  onOpenChange,
  onSaved,
}: {
  open: boolean;
  onOpenChange: (o: boolean) => void;
  onSaved?: () => void;
}) {
  const { t, lang } = useT();
  const baseCurrency = useSettings(s => s.baseCurrency);

  const [amount, setAmount] = useState('');
  const [fromAccountId, setFromAccountId] = useState('');
  const [toAccountId, setToAccountId] = useState('');
  const [date, setDate] = useState('');
  const [note, setNote] = useState('');
  const [accounts, setAccounts] = useState<Account[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (open) {
      api<{ data: Account[] }>('/api/accounts').then(a => {
        setAccounts(a.data);
        const def = a.data.find(x => x.isDefault) || a.data[0];
        setFromAccountId(def?.id || '');
        setToAccountId(a.data.find(x => x.id !== def?.id)?.id || '');
      });
      setDate(new Date().toISOString().slice(0, 10));
      setAmount('');
      setNote('');
      setError(null);
    }
  }, [open]);

  const fromAccount = accounts.find(a => a.id === fromAccountId);
  const toAccount = accounts.find(a => a.id === toAccountId);

  // Currency conversion preview
  const convertedAmount = (() => {
    const amt = parseFloat(amount);
    if (!amt || !fromAccount || !toAccount) return null;
    if (fromAccount.currency === toAccount.currency) return amt;
    return convert(amt, fromAccount.currency as Currency, toAccount.currency as Currency);
  })();

  const handleSwap = () => {
    const tmp = fromAccountId;
    setFromAccountId(toAccountId);
    setToAccountId(tmp);

export function Providers({ children }: { children: React.ReactNode }) {
  const lang = useSettings(s => s.lang);
  const theme = useSettings(s => s.theme);

  useEffect(() => {
    const html = document.documentElement;
    html.lang = lang;
    html.dir = lang === 'ar' ? 'rtl' : 'ltr';
  }, [lang]);

  useEffect(() => {
    const html = document.documentElement;
    const apply = (t: 'light' | 'dark') => {
      if (t === 'dark') html.classList.add('dark');
      else html.classList.remove('dark');
    };
    if (theme === 'system') {
      const mq = window.matchMedia('(prefers-color-scheme: dark)');
      apply(mq.matches ? 'dark' : 'light');
      const handler = (e: MediaQueryListEvent) => apply(e.matches ? 'dark' : 'light');
      mq.addEventListener('change', handler);
      return () => mq.removeEventListener('change', handler);
    }
    apply(theme);
  }, [theme]);

  return (
    <>
      <ServiceWorkerRegistrar />
      {children}
    </>
  );
}

'use client';
import { useEffect } from 'react';
import { useSettings } from '@/lib/store';
import { toast } from 'sonner';

export function ServiceWorkerRegistrar() {
  const lang = useSettings(s => s.lang);

  useEffect(() => {
    if ('serviceWorker' in navigator) {
      // Unregister all existing service workers first (to clear old caches)
      navigator.serviceWorker.getRegistrations().then((regs) => {
        const old = regs.filter(r => r.active && !r.active.scriptURL.endsWith('/sw.js'));
        // If the active SW is from a previous version, force unregister to refresh
        return Promise.all(old.map(r => r.unregister()));
      }).then(() => {
        // Register the new SW
        return navigator.serviceWorker.register('/sw.js');
      }).then((reg) => {
        console.log('SW registered v3');
        // Force update if there's a new version
        if (reg.waiting) {
          reg.waiting.postMessage('SKIP_WAITING');
        }
        reg.addEventListener('updatefound', () => {
          const nw = reg.installing;
          if (nw) {
            nw.addEventListener('statechange', () => {
              if (nw.state === 'activated') {
                toast.info(lang === 'ar' ? 'تم تحديث التطبيق' : 'App updated');
                setTimeout(() => window.location.reload(), 1000);
              }
            });
          }
        });
      }).catch((err) => {
        console.warn('SW failed:', err);
      });
    }
  }, [lang]);

  useEffect(() => {
    const updateOnlineStatus = () => {
      if (!navigator.onLine) {
        toast.warning(lang === 'ar' ? 'أنت غير متصل - وضع عدم الاتصال' : 'Offline mode');
      }
    };
    window.addEventListener('online', updateOnlineStatus);
    window.addEventListener('offline', updateOnlineStatus);
    return () => {
      window.removeEventListener('online', updateOnlineStatus);
      window.removeEventListener('offline', updateOnlineStatus);
    };
  }, [lang]);

  return null;
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { verifyPassword, generateToken } from '@/lib/auth';

// POST /api/auth/login
// Body: { email, password }
export async function POST(req: NextRequest) {
  try {
    const { email, password } = await req.json();

    if (!email || !password) {
      return NextResponse.json({ error: 'Email and password required' }, { status: 400 });
    }

    const user = await db.user.findUnique({ where: { email } });
    if (!user) {
      return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 });
    }

    if (!verifyPassword(password, user.passwordHash)) {
      return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 });
    }

    const token = generateToken(user.id);

    const response = NextResponse.json({
      data: { id: user.id, email: user.email, name: user.name, token },
    });

    response.cookies.set('auth-token', token, {
      httpOnly: true,
      sameSite: 'lax',
      maxAge: 30 * 24 * 60 * 60,
      path: '/',
    });

    return response;
  } catch (err: any) {
    console.error('Login error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

// GET /api/stats?month=YYYY-MM&baseCurrency=EGP
export async function GET(req: NextRequest) {
  try {
    const url = new URL(req.url);
    const month = url.searchParams.get('month');
    const baseCurrency = (url.searchParams.get('baseCurrency') || 'EGP') as Currency;

    let start: Date;
    let end: Date;
    if (month) {
      start = new Date(`${month}-01T00:00:00.000Z`);
      end = new Date(start);
      end.setMonth(end.getMonth() + 1);
    } else {
      start = new Date();
      start.setDate(1);
      start.setHours(0, 0, 0, 0);
      end = new Date(start);
      end.setMonth(end.getMonth() + 1);
    }

    const transactions = await db.transaction.findMany({
      where: { date: { gte: start, lt: end } },
      include: { category: true, account: true },
    });

    const accounts = await db.account.findMany();

    // Helper: convert from tx currency to base currency
    const toBase = (amount: number, from: string) => {
      const fromCur = (from as Currency) || 'EGP';
      if (fromCur === baseCurrency) return amount;
      const inEgp = amount * EXCHANGE_RATES[fromCur];
      return inEgp / EXCHANGE_RATES[baseCurrency];
    };

    // Exclude transfers (both TRANSFER type AND "تحويلات" category from old Money Manager import)
    const isTransfer = (t: any) =>
      t.type === 'TRANSFER' ||
      t.category?.nameAr === 'تحويلات' ||
      t.category?.nameEn === 'Transfers';

    const expenses = transactions.filter(t => t.type === 'EXPENSE' && !isTransfer(t));
    const incomes = transactions.filter(t => t.type === 'INCOME' && !isTransfer(t));
    const transfers = transactions.filter(t => isTransfer(t));

    const totalExpense = expenses.reduce((s, t) => s + toBase(t.amount, t.currency), 0);
    const totalIncome = incomes.reduce((s, t) => s + toBase(t.amount, t.currency), 0);
    const totalTransfers = transfers.reduce((s, t) => s + toBase(t.amount, t.currency), 0);
    const balance = totalIncome - totalExpense;

    // total assets = sum of account balances converted to base
    const totalAssets = accounts.reduce(
      (s, a) => s + toBase(a.balance, a.currency),
      0
    );

    // تجميع حسب الفئة (مع تحويل)
    const byCategoryMap: Record<string, { total: number; count: number; category: any }> = {};
    expenses.forEach(t => {
      const cid = t.categoryId;
      if (!byCategoryMap[cid]) byCategoryMap[cid] = { total: 0, count: 0, category: t.category };
      byCategoryMap[cid].total += toBase(t.amount, t.currency);
      byCategoryMap[cid].count += 1;
    });

    // تجميع يومي للـ chart (مع تحويل)
    const daysInMonth = new Date(end.getFullYear(), end.getMonth(), 0).getDate();
    const byDay: { date: string; expense: number; income: number }[] = [];
    for (let day = 1; day <= daysInMonth; day++) {
      const dateStr = `${month || start.toISOString().slice(0, 7)}-${String(day).padStart(2, '0')}`;
      byDay.push({ date: dateStr, expense: 0, income: 0 });
    }
    transactions.forEach(t => {
      const day = t.date.getDate();
      const idx = day - 1;
      if (idx >= 0 && idx < byDay.length) {
        if (t.type === 'EXPENSE') byDay[idx].expense += toBase(t.amount, t.currency);
        else byDay[idx].income += toBase(t.amount, t.currency);
      }
    });

    return NextResponse.json({
      data: {
        totalExpense,
        totalIncome,
        totalTransfers,
        balance,
        totalAssets,
        baseCurrency,
        expenseCount: expenses.length,
        incomeCount: incomes.length,
        transferCount: transfers.length,
        byCategory: Object.values(byCategoryMap).sort((a: any, b: any) => b.total - a.total),
        byDay,
        month: month || start.toISOString().slice(0, 7),
      },
    });
  } catch (err: any) {
    console.error('GET /api/stats error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { convert, type Currency } from '@/lib/currency';

/**
 * POST /api/transfer
 * Body: { fromAccountId, toAccountId, amount, currency, date, note }
 *
 * Creates a transfer between two accounts:
 *  1. Records one TRANSFER transaction with accountId=source, toAccountId=destination
 *  2. Updates both account balances (source -= amount, destination += amount)
 *  3. Currency conversion happens automatically if accounts have different currencies
 */
export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { fromAccountId, toAccountId, amount, currency, date, note } = body;

    if (!fromAccountId || !toAccountId || !amount || fromAccountId === toAccountId) {
      return NextResponse.json(
        { error: 'fromAccountId, toAccountId, amount are required and must be different' },
        { status: 400 }
      );
    }

    const amt = Number(amount);
    if (amt <= 0) {
      return NextResponse.json({ error: 'Amount must be positive' }, { status: 400 });
    }

    // Get both accounts
    const [sourceAccount, destAccount] = await Promise.all([
      db.account.findUnique({ where: { id: fromAccountId } }),
      db.account.findUnique({ where: { id: toAccountId } }),
    ]);

    if (!sourceAccount) return NextResponse.json({ error: 'Source account not found' }, { status: 404 });
    if (!destAccount) return NextResponse.json({ error: 'Destination account not found' }, { status: 404 });

    // Currency conversion
    const txCurrency = (currency || sourceAccount.currency) as Currency;
    const sourceAmountInSourceCur =
      txCurrency === sourceAccount.currency
        ? amt
        : convert(amt, txCurrency, sourceAccount.currency as Currency);
    const destAmountInDestCur =
      txCurrency === destAccount.currency
        ? amt
        : convert(amt, txCurrency, destAccount.currency as Currency);

    // Update account balances
    await db.account.update({
      where: { id: fromAccountId },
      data: { balance: sourceAccount.balance - sourceAmountInSourceCur },
    });
    await db.account.update({
      where: { id: toAccountId },
      data: { balance: destAccount.balance + destAmountInDestCur },
    });

    // Get or create the "تحويلات" category for transfers
    let transferCat = await db.category.findFirst({
      where: { nameAr: 'تحويلات', type: 'TRANSFER' as any },
    });
    if (!transferCat) {
      // Fall back to existing "تحويلات" or create new
      transferCat = await db.category.findFirst({ where: { nameAr: 'تحويلات' } });
      if (!transferCat) {
        transferCat = await db.category.create({
          data: {
            nameAr: 'تحويلات',
            nameEn: 'Transfers',
            type: 'EXPENSE' as any, // Will be unused since tx type is TRANSFER
            icon: 'arrow-right-left',
            color: '#64748B',
          },
        });
      }
    }

    // Create ONE transfer transaction
    const transfer = await db.transaction.create({
      data: {
        type: 'TRANSFER' as any,
        amount: amt,
        currency: txCurrency,
        accountId: fromAccountId,
        categoryId: transferCat.id,
        toAccountId: toAccountId,
        date: date ? new Date(date) : new Date(),
        note: note || `تحويل من ${sourceAccount.name} إلى ${destAccount.name}`,
      },
      include: {
        account: true,
        category: true,
        toAccount: true,
      },
    });

    return NextResponse.json({ data: transfer });
  } catch (err: any) {
    console.error('Transfer error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

          categoryName = txType === 'INCOME' ? 'دخل إضافي' : 'مصروفات أخرى';
        }

        // Note - combine Note + Description
        const noteText = [
          noteKey ? String(row[noteKey] || '').trim() : '',
          descKey ? String(row[descKey] || '').trim() : '',
        ].filter(Boolean).join(' • ');
        const finalNote = rowType === 'transfer-out' && catName
          ? `${noteText} → ${catName}`.trim()
          : noteText;

        if (!account) { skipped++; continue; }

        const category = await ensureCategory(categoryName, rowType, amount);
        if (!category) { skipped++; continue; }

          where: { id: account.id },
          data: { balance: { increment: delta } },

            type: txType,
            amount: absAmount,
            currency: validCur,
            accountId: account.id,
            categoryId: category.id,

          },
        });

      } catch (e: any) {
        errors.push(`Row error: ${e.message}`);
        if (errors.length > 20) break;
      }
    }

    return NextResponse.json({

      imported,
      skipped,
      errors: errors.slice(0, 10),
      message: `✅ Imported ${imported} transactions${skipped ? `, skipped ${skipped}` : ''}${errors.length ? `, ${errors.length} errors` : ''}`,

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';

// GET /api/backup/export - ينزّل ملف JSON بكل البيانات
export async function GET() {
  try {
    const [accounts, categories, transactions, budgets, settings] = await Promise.all([
      db.account.findMany({ include: { transactions: true } }),
      db.category.findMany({ include: { transactions: true, budgets: true } }),
      db.transaction.findMany({ include: { account: true, category: true } }),
      db.budget.findMany({ include: { category: true } }),
      db.setting.findMany(),
    ]);

    const backup = {
      version: 1,
      app: 'mywallet',
      exportedAt: new Date().toISOString(),
      data: {
        accounts: accounts.map(({ transactions: _txs, ...rest }) => rest),
        categories: categories.map(({ transactions: _txs, budgets: _bs, ...rest }) => rest),
        transactions: transactions.map(({ account: _a, category: _c, ...rest }) => ({
          ...rest,
          accountName: _a?.name,
          categoryName: _c?.nameAr,
        })),
        budgets: budgets.map(({ category: _c, ...rest }) => ({
          ...rest,
          categoryName: _c?.nameAr,
        })),
        settings,
      },
    };

    const json = JSON.stringify(backup, null, 2);

    return new NextResponse(json, {
      headers: {
        'Content-Type': 'application/json; charset=utf-8',
        'Content-Disposition': `attachment; filename="mywallet-backup-${new Date().toISOString().slice(0, 10)}.json"`,
      },
    });
  } catch (err: any) {
    console.error('Backup export error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

// POST /api/backup/import - يستورد ملف JSON
export async function POST(req: NextRequest) {
  try {
    const formData = await req.formData();
    const file = formData.get('file') as File;
    if (!file) {
      return NextResponse.json({ error: 'No file uploaded' }, { status: 400 });
    }

    const text = await file.text();
    const backup = JSON.parse(text);

    if (!backup?.data || !backup.data.accounts || !backup.data.categories) {
      return NextResponse.json({ error: 'Invalid backup file format' }, { status: 400 });
    }

    const { accounts, categories, transactions, budgets, settings } = backup.data;

    // Clean current data ( Cascading deletes for transactions)
    await db.transaction.deleteMany({});
    await db.budget.deleteMany({});
    await db.account.deleteMany({});
    await db.category.deleteMany({});
    await db.setting.deleteMany({});

    // restore categories first
    let importedCategories = 0;
    for (const c of categories || []) {
      await db.category.create({
        data: {
          id: c.id,
          nameAr: c.nameAr,
          nameEn: c.nameEn,
          type: c.type,
          icon: c.icon || 'tag',
          color: c.color || '#10B981',
          isDefault: c.isDefault || false,
        },
      });
      importedCategories++;
    }

    // restore accounts
    let importedAccounts = 0;
    for (const a of accounts || []) {
      await db.account.create({
        data: {
          id: a.id,
          name: a.name,
          type: a.type,
          currency: a.currency,
          balance: a.balance,
          color: a.color || '#2563EB',
          icon: a.icon || 'wallet',
          isDefault: a.isDefault || false,
        },
      });
      importedAccounts++;
    }

    // restore transactions
    let importedTransactions = 0;
    for (const t of transactions || []) {
      await db.transaction.create({
        data: {
          id: t.id,
          type: t.type,
          amount: t.amount,
          currency: t.currency,
          accountId: t.accountId,
          categoryId: t.categoryId,
          date: new Date(t.date),
          note: t.note,
          tags: t.tags,
        },
      });
      importedTransactions++;
    }

    // restore budgets
    let importedBudgets = 0;
    for (const b of budgets || []) {
      await db.budget.create({
        data: {
          id: b.id,
          categoryId: b.categoryId,
          month: b.month,
          limit: b.limit,
          currency: b.currency || 'EGP',
        },
      });
      importedBudgets++;
    }

    // restore settings
    let importedSettings = 0;
    for (const s of settings || []) {
      await db.setting.create({
        data: {
          key: s.key,
          value: s.value,
        },
      });
      importedSettings++;
    }

    return NextResponse.json({
      success: true,
      imported: {
        accounts: importedAccounts,
        categories: importedCategories,
        transactions: importedTransactions,
        budgets: importedBudgets,
        settings: importedSettings,
      },
    });
  } catch (err: any) {
    console.error('Backup import error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';

// POST /api/reset - Delete ALL data (transactions, budgets, accounts, categories)
export async function POST() {
  try {
    // Delete in order (respecting foreign keys)
    const txCount = await db.transaction.count();
    const budgetCount = await db.budget.count();
    const recurringCount = await db.recurringTransaction?.count?.() || 0;
    const savingsCount = await db.savingsGoal?.count?.() || 0;
    const billCount = await db.billReminder?.count?.() || 0;

    await db.transaction.deleteMany({});
    await db.budget.deleteMany({});
    
    // Delete optional tables if they exist
    try { await (db as any).recurringTransaction.deleteMany({}); } catch {}
    try { await (db as any).savingsGoal.deleteMany({}); } catch {}
    try { await (db as any).billReminder.deleteMany({}); } catch {}

    // Don't delete accounts and categories - keep them for reuse
    // But reset account balances to 0
    const accounts = await db.account.findMany();
    for (const acc of accounts) {
      await db.account.update({
        where: { id: acc.id },
        data: { balance: 0 },
      });
    }

    return NextResponse.json({
      success: true,
      deleted: {
        transactions: txCount,
        budgets: budgetCount,
        recurring: recurringCount,
        savingsGoals: savingsCount,
        billReminders: billCount,
      },
      message: `Deleted ${txCount} transactions, ${budgetCount} budgets`,
    });
  } catch (err: any) {
    console.error('Reset error:', err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';

// GET /api/budgets?month=YYYY-MM
export async function GET(req: NextRequest) {
  try {
    const url = new URL(req.url);
    const month = url.searchParams.get('month');
    const where: any = {};
    if (month) where.month = month;

    const budgets = await db.budget.findMany({
      where,
      include: { category: true },
    });

    // اجمع المنصرف الفعلي لكل فئة في الشهر
    const startOfMonth = new Date(`${month}-01T00:00:00.000Z`);
    const endOfMonth = new Date(startOfMonth);
    endOfMonth.setMonth(endOfMonth.getMonth() + 1);

    const spent = await db.transaction.groupBy({
      by: ['categoryId'],
      where: {
        type: 'EXPENSE',
        date: { gte: startOfMonth, lt: endOfMonth },
      },
      _sum: { amount: true },
    });

    const spentMap = new Map(spent.map(s => [s.categoryId, s._sum.amount || 0]));

    return NextResponse.json({
      data: budgets.map(b => ({
        ...b,
        spent: spentMap.get(b.categoryId) || 0,
      })),
    });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

// POST /api/budgets
export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const { categoryId, month, limit, currency } = body;
    if (!categoryId || !month || !limit) {
      return NextResponse.json(
        { error: 'categoryId, month, limit are required' },
        { status: 400 }
      );
    }

    // upsert: لو موجود، حدّث الحد
    const existing = await db.budget.findUnique({
      where: { categoryId_month: { categoryId, month } },
    });

    if (existing) {
      const updated = await db.budget.update({
        where: { id: existing.id },
        data: { limit: Number(limit), currency: currency || existing.currency },
        include: { category: true },
      });
      return NextResponse.json({ data: updated });
    }

    const created = await db.budget.create({
      data: { categoryId, month, limit: Number(limit), currency: currency || 'EGP' },
      include: { category: true },
    });
    return NextResponse.json({ data: created });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
