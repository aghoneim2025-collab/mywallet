  هذه المهلة."

  Response: this is correct and intentional. The NEG_CACHE_TTL_MS is set
  to 1 second specifically to coalesce concurrent parallel requests (the
  home screen fires 3+ fetches at once on mount, all with the same token).
  Without this short cache, the DB would be hit 3+ times per page load.

  The MAXIMUM cross-worker revocation delay is 1 second (NEG_CACHE_TTL_MS).
  After 1 second, even a worker that previously cached "not revoked" will
  re-fetch from the DB and see the new revocation row.

  For tighter revocation (0 second delay), set NEG_CACHE_TTL_MS = 0 — every
  request does a DB lookup. Trade-off: more DB load + higher latency per
  request. For a personal finance app with one user at a time, the 1s
  window is acceptable.

/**
 * Secure authentication - bcrypt + HMAC-signed tokens + server-side revocation.
 *
 * History:
 *   - P0.1: Math.random() per call → tokens always invalid in next request
 *   - P0.2: randomBytes(32) at module load → still broken because Next.js dev
 *           runs MULTIPLE worker processes, each with its own randomBytes
 *   - P0.4: deterministic dev secret (string-based, shared across workers)
 *   - P0.5: server-side token revocation via DB-backed blacklist
 *   - P0.6: store SHA-256 hash of token (not raw) in DB blacklist, so a DB
 *           leak doesn't turn the table into a vault of usable tokens.
 *           Rotate AUTH_SECRET after every leak; new secret invalidates ALL
 *           outstanding tokens (their HMAC signature no longer matches).
 */
import bcrypt from 'bcryptjs';
import { createHmac, timingSafeEqual, createHash } from 'crypto';
import { db } from './db';

const TOKEN_SECRET = process.env.AUTH_SECRET;
// P0.9: only allow DEV_SECRET when NODE_ENV is EXACTLY 'development'.
// Previously checked `isProduction` (NODE_ENV === 'production'), which meant
// that 'test', 'staging', or undefined NODE_ENV would silently fall back
// to the publicly-known DEV_SECRET. Now: any NODE_ENV other than
// 'development' or 'production' is treated as production-grade strict.
const nodeEnv = process.env.NODE_ENV;
const isDev = nodeEnv === 'development';
const isProd = nodeEnv === 'production' || (!isDev);  // treat unknown as prod
// Keep backward-compat name for existing imports:
// P0.13: exported for use by logout route (to know which cookie to delete)
export const isProduction = isProd;

// P0.4: Stable, deterministic dev secret so all workers share the same key.
const DEV_SECRET = 'dev-only-secret-please-set-AUTH_SECRET-in-production-env-min-32-chars-' +
  'do-not-use-in-prod-' + (nodeEnv || 'development');

// P0.8: known-bad placeholder values from .env.example.
const FORBIDDEN_SECRETS = new Set([
  'REPLACE_ME', 'replace-me', 'replace_me', 'changeme', 'change-me',
  'placeholder', 'todo', 'xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx',
  'dev-only-secret-please-set-AUTH_SECRET-in-production-env-min-32-chars-do-not-use-in-prod-development',
]);

function isPlaceholderSecret(s: string | undefined): boolean {
  if (!s) return false;
  if (FORBIDDEN_SECRETS.has(s)) return true;
  const lower = s.toLowerCase();
  if (lower.startsWith('replace') && lower.length < 64) return true;
  if (lower.startsWith('changeme')) return true;
  if (lower.startsWith('todo')) return true;
  if (lower.startsWith('placeholder')) return true;
  return false;
}

// P0.9: error class to distinguish DB outages from auth failures.
// Thrown by isTokenRevoked when the DB lookup fails. Callers should
// catch this and return 503 (Service Unavailable), NOT 401 — the user's
// token may be perfectly valid, the DB is just temporarily unreachable.
export class DbUnavailableError extends Error {
  constructor(message: string, public cause?: unknown) {
    super(message);
    this.name = 'DbUnavailableError';
  }
}

// P0.9: errors that mean the user's auth is genuinely invalid (revoked,
// bad signature, expired). Distinct from DB outages.
export class AuthInvalidError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'AuthInvalidError';
  }
}

// Eagerly validate at module load time.
// P0.8: rejects placeholder values even if they meet length requirement.
// P0.9: applies whenever NODE_ENV is NOT 'development' (covers prod,
// test, staging, undefined, etc.). Only 'development' allows DEV_SECRET.
if (isProd && (!TOKEN_SECRET || TOKEN_SECRET.length < 32 || isPlaceholderSecret(TOKEN_SECRET))) {
  throw new Error(
    'AUTH_SECRET must be set to a random string of at least 32 characters when NODE_ENV is not "development". ' +
    'Current NODE_ENV: ' + JSON.stringify(nodeEnv) + '. ' +
    'Generate one with: openssl rand -hex 32. ' +
    (isPlaceholderSecret(TOKEN_SECRET) ? 'Placeholder values are not allowed.' : '')
  );
}

function getSecret(): string {
  if (isProd) return TOKEN_SECRET!;
  // Dev mode: use AUTH_SECRET if set + non-placeholder, else DEV_SECRET.
  return TOKEN_SECRET && TOKEN_SECRET.length >= 32 && !isPlaceholderSecret(TOKEN_SECRET)
    ? TOKEN_SECRET
    : DEV_SECRET;
}

const TOKEN_EXPIRY_MS = 30 * 24 * 60 * 60 * 1000; // 30 days

// P0.6: SHA-256 hash the token before storing/looking up in DB or cache.
// This way, a DB dump doesn't expose usable tokens — only hashes that
// can't be reversed back to the original HMAC-signed token.
//
// The cache is keyed by the SAME hash so that a positive cache hit means
// the token (whose hash we know) is revoked.
export function hashToken(token: string): string {
  return createHash('sha256').update(token).digest('hex');
}

// P0.5: in-memory cache for "definitely revoked" tokens.
// P0.6: keyed by SHA-256 hash, NOT the raw token.
// We ONLY cache positive revocation results (token IS revoked).
// We do NOT cache negative results ("not revoked") for long because that
// would delay revocation by up to the cache TTL.
//
// Set of revoked-token hashes (cleared on cleanup).
const revokedTokenSet = new Set<string>();

// P0.5: optional short-TTL negative cache to reduce DB load on hot paths.
// Set to 0 to disable (every request hits DB). 1000ms = max 1s revocation delay.
// P0.6: keyed by SHA-256 hash, NOT the raw token.
//
// IMPORTANT NOTE about multi-worker revocation latency:
// This negative cache is per-worker. If worker A served the logout (clearing
// its negative cache and adding the token to its positive cache), worker B
// might still have the token in its negative cache for up to 1 second.
// During that window, worker B may serve a request with the revoked token.
// The MAXIMUM revocation delay across workers = NEG_CACHE_TTL_MS (1 second).
const NEG_CACHE_TTL_MS = 1000; // 1 second — minimal cross-worker revocation window
const notRevokedCache = new Map<string, number>(); // hash -> fetchedAt timestamp

// P0.6/P0.7: time-based cleanup schedule (in addition to count-based).
// Every CLEANUP_INTERVAL_MS, the next logout triggers cleanup.
// Also runs at most once per CLEANUP_INTERVAL_MS regardless of logout count.
// This ensures cleanup runs even if logouts are infrequent.
const CLEANUP_INTERVAL_MS = 60 * 60 * 1000; // 1 hour
let lastCleanupAt = 0;

// P0.6: opportunistic cleanup counter — also runs every Nth logout.
const CLEANUP_EVERY_N_LOGOUTS = 10;
let logoutCounter = 0;

export async function hashPassword(password: string): Promise<string> {
  const salt = await bcrypt.genSalt(12);
  return bcrypt.hash(password, salt);
}

export async function verifyPassword(password: string, hash: string): Promise<boolean> {
  return bcrypt.compare(password, hash);
}

export function generateToken(userId: string): string {
  const payload = { userId, exp: Date.now() + TOKEN_EXPIRY_MS };
  const payloadStr = JSON.stringify(payload);
  const payloadB64 = Buffer.from(payloadStr).toString('base64url');
  const signature = createHmac('sha256', getSecret()).update(payloadB64).digest('base64url');
  return `${payloadB64}.${signature}`;
}

// P0.5: Extract token from request (Authorization header OR cookie).
// P0.12: in production, cookie name is `__Host-auth-token` (with prefix).
export function extractToken(req: Request): string | null {
  // Authorization: Bearer header (used by some legacy clients + tests)
  const authHeader = req.headers.get('authorization');
  if (authHeader?.startsWith('Bearer ')) {
    return authHeader.slice(7);
  }
  // Cookie (primary auth path now — JWT no longer in localStorage)
  const cookie = req.headers.get('cookie') || '';
  // Match either `auth-token` (dev) or `__Host-auth-token` (prod)
  const tokenMatch = cookie.match(/(?:__Host-)?auth-token=([^;]+)/);
  if (tokenMatch) return tokenMatch[1];
  return null;
}

// P0.5: Synchronous verify (HMAC + expiry only). Does NOT check blacklist.
// Use this only when you need to extract the userId quickly without DB.
export function verifyTokenSignature(token: string): { userId: string; exp: number } | null {
  try {
    const parts = token.split('.');
    if (parts.length !== 2) return null;
    const [payloadB64, signature] = parts;
    if (!payloadB64 || !signature) return null;

    const expectedSig = createHmac('sha256', getSecret()).update(payloadB64).digest('base64url');
    const sigBuf = Buffer.from(signature);
    const expBuf = Buffer.from(expectedSig);
    if (sigBuf.length !== expBuf.length) return null;
    if (!timingSafeEqual(sigBuf, expBuf)) return null;

    const payload = JSON.parse(Buffer.from(payloadB64, 'base64url').toString());
    if (typeof payload.userId !== 'string' || !payload.userId) return null;
    if (typeof payload.exp !== 'number' || payload.exp < Date.now()) return null;
    return { userId: payload.userId, exp: payload.exp };
  } catch {
    return null;
  }
}

// P0.8/P0.9: fail-closed BUT distinguish DB outage from revoked token.
// - If DB lookup throws → throw DbUnavailableError. Callers should
//   return 503 (Service Unavailable), NOT 401. The user's token may be
//   perfectly valid; the DB is just temporarily unreachable.
// - If DB lookup succeeds and finds the token in the blacklist → return true.
// - If DB lookup succeeds and doesn't find it → return false + cache negatively.
//
// Why throw instead of return true: returning true silently makes it
// impossible for callers (session route, requireAuthAsync) to distinguish
// "token is revoked" from "DB is down". With a throw, callers can:
//   - /api/auth/session: catch → return 503, frontend keeps token, retries
//   - requireAuthAsync: catch → return 503 (not 401), so the user knows
//     to retry rather than re-login
//
// This is still "fail-closed" in the security sense: a DB outage blocks
// access rather than allowing a stolen token through. But it's also
// honest about WHY access is blocked.
async function isTokenRevoked(token: string): Promise<boolean> {
  const tokenHash = hashToken(token);
  if (revokedTokenSet.has(tokenHash)) return true;
  const negCacheEntry = notRevokedCache.get(tokenHash);
  if (negCacheEntry && (Date.now() - negCacheEntry) < NEG_CACHE_TTL_MS) {
    return false;
  }
  // P0.9: throw DbUnavailableError on DB failure. Callers distinguish this
  // from "token is revoked" and can return 503 instead of 401.
  let found;
  try {
    found = await db.revokedToken.findUnique({ where: { token: tokenHash }, select: { id: true } });
  } catch (e) {
    console.error('isTokenRevoked: DB lookup failed, throwing DbUnavailableError:', e);
    throw new DbUnavailableError('Token revocation check failed: database unavailable', e);
  }
  if (found) {
    revokedTokenSet.add(tokenHash);
    notRevokedCache.delete(tokenHash);
    return true;
  }
  notRevokedCache.set(tokenHash, Date.now());
  return false;
}

// P0.8: Revoke a token. THROWS on DB error (does not swallow).
// Callers (logout route) must propagate the error to the user, so they
// know revocation failed and can retry. Otherwise a stolen token would
// remain valid while the user thinks they've logged out.
export async function revokeToken(token: string, userId: string): Promise<void> {
  const verified = verifyTokenSignature(token);
  if (!verified) return; // can't revoke an invalid token — nothing to revoke
  const tokenHash = hashToken(token);
  // P0.8: let errors propagate. Caller MUST handle.
  await db.revokedToken.upsert({
    where: { token: tokenHash },
    create: { token: tokenHash, userId, expiresAt: new Date(verified.exp) },
    update: {}, // already revoked
  });
  // Update local caches only AFTER successful DB write
  revokedTokenSet.add(tokenHash);
  notRevokedCache.delete(tokenHash);
}

// P0.5/P0.6: Cleanup of expired revoked tokens.
// Called opportunistically on logout — every CLEANUP_EVERY_N_LOGOUTS (10) logouts.
// Also clears the in-memory revokedTokenSet so it doesn't grow forever.
//
// For a more aggressive schedule, this could be wired to a cron job in production.
export async function cleanupExpiredRevokedTokens(): Promise<number> {
  try {
    const result = await db.revokedToken.deleteMany({
      where: { expiresAt: { lt: new Date() } },
    });
    if (result.count > 0) {
      // Clear local positive cache so it doesn't grow forever.
      // (Next request for a still-valid token will re-fetch from DB.)
      revokedTokenSet.clear();
      notRevokedCache.clear();
    }
    return result.count;
  } catch {
    return 0;
  }
}

// P0.7: Called from /api/auth/logout to opportunistically trigger cleanup.
// Runs cleanup in TWO cases:
//   (1) every CLEANUP_EVERY_N_LOGOUTS (10) logouts — count-based
//   (2) if last cleanup was more than CLEANUP_INTERVAL_MS (1 hour) ago — time-based
// The time-based path ensures cleanup runs even if logouts are infrequent.
// Returns the cleanup count (for logging) or 0 if cleanup was skipped this round.
export async function maybeCleanupExpiredRevokedTokens(): Promise<number> {
  logoutCounter++;
  const now = Date.now();
  const timeSinceLastCleanup = now - lastCleanupAt;
  const shouldRun =
    (logoutCounter % CLEANUP_EVERY_N_LOGOUTS === 0) ||
    (timeSinceLastCleanup >= CLEANUP_INTERVAL_MS);
  if (shouldRun) {
    lastCleanupAt = now;
    return cleanupExpiredRevokedTokens();
  }
  return 0;
}

export function verifyToken(token: string): { userId: string } | null {
  const result = verifyTokenSignature(token);
  return result ? { userId: result.userId } : null;
}

// P0.5: Async version that also checks the revocation blacklist.
// P0.9: throws DbUnavailableError when DB lookup fails (caller should
// return 503, not 401). Returns null when token is missing/invalid/revoked.
export async function getUserIdFromRequestAsync(req: Request): Promise<string | null> {
  const token = extractToken(req);
  if (!token) return null;
  const verified = verifyTokenSignature(token);
  if (!verified) return null;  // bad signature, expired, or malformed
  // P0.9: isTokenRevoked throws DbUnavailableError on DB failure.
  // We let it propagate so callers can distinguish auth-invalid from
  // service-unavailable.
  if (await isTokenRevoked(token)) return null;
  return verified.userId;
}

// Synchronous version (kept for backwards compat, but does NOT check blacklist).
export function getUserIdFromRequest(req: Request): string | null {
  const token = extractToken(req);
  if (!token) return null;
  const result = verifyTokenSignature(token);
  return result ? result.userId : null;
}

// P0.5: Async requireAuth that checks blacklist.
// P0.9: throws DbUnavailableError on DB failure. Callers should catch
// and return 503 (Service Unavailable) — the user's token may be valid,
// the DB is just down.
export async function requireAuthAsync(req: Request): Promise<string> {
  const userId = await getUserIdFromRequestAsync(req);
  if (!userId) throw new AuthInvalidError('Unauthorized');
  return userId;
}

// Synchronous requireAuth (legacy, doesn't check blacklist — used only for
// GET endpoints where stale auth for ~30s is acceptable).
export function requireAuth(req: Request): string {
  const userId = getUserIdFromRequest(req);
  if (!userId) throw new Error('Unauthorized');
  return userId;
}

export function setAuthCookie(response: Response, token: string): Response {
  // P0.12: harden cookie attributes.
  // - __Host- prefix: cookie name must start with `__Host-` to prevent
  //   subdomain cookie injection. Requires Secure + Path=/ + no Domain.
  // - SameSite=Strict: cookie is never sent on cross-site requests
  //   (was: SameSite=Lax which allows top-level navigations).
  // - Secure: cookie only sent over HTTPS (always set in production).
  // - HttpOnly: cookie not readable by JS (XSS can't steal it).
  //
  // Trade-off: SameSite=Strict means clicking a link from a third-party
  // site to /api/excel?action=export won't send the cookie → user has to
  // be on the app's domain. For a finance app, this is the correct trade-off.
  const secure = isProduction ? '; Secure' : '';
  const hostPrefix = isProduction ? '__Host-' : '';
  response.headers.set(
    'Set-Cookie',
    `${hostPrefix}auth-token=${token}; HttpOnly; SameSite=Strict${secure}; Max-Age=2592000; Path=/`,
  );
  return response;
}

export function formatMoney(
  amount: number,
  currency: Currency,
  lang: 'ar' | 'en' = 'ar'
): string {
  const sym = CURRENCY_SYMBOLS[currency][lang];
  const formatted = new Intl.NumberFormat(lang === 'ar' ? 'ar-EG' : 'en-US', {

  const sign = amount < 0 ? '-' : '';
  return lang === 'ar'
    ? `${sign}${formatted} ${sym}`
    : `${sign}${sym} ${formatted}`;
}

    notation: 'compact',
    maximumFractionDigits: 1,
  }).format(amount);
}

// P0.4/P0.12: simple fetch wrapper.
//
// P0.12 changes: no longer attaches `Authorization: Bearer <token>` header.
// The JWT now lives in the HttpOnly cookie (set by /api/auth/login +
// /api/auth/register). The browser automatically sends cookies for
// same-origin requests, so no client-side intervention is needed.
//
// `credentials: 'same-origin'` is the default for fetch in modern browsers,
// which is what we want — cookies are sent on same-origin requests only.
//
// Future: if we ever add cross-origin API access (e.g., a mobile app),
// add `credentials: 'include'` AND ensure the server sets
// `Access-Control-Allow-Credentials: true` + an explicit origin (NOT *).

export async function api<T>(
  path: string,
  opts?: RequestInit & { params?: Record<string, string | undefined> }
): Promise<T> {
  const { params, ...rest } = opts || {};
  let url = path;
  if (params) {
    const sp = new URLSearchParams();
    Object.entries(params).forEach(([k, v]) => {
      if (v !== undefined) sp.set(k, v);
    });
    const qs = sp.toString();
    if (qs) url += `?${qs}`;
  }

  const res = await fetch(url, {
    ...rest,
    headers: { 'Content-Type': 'application/json', ...(rest?.headers || {}) },
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: 'Network error' }));
    throw new Error(err.error || res.statusText);
  }

  return res.json();
}

// Helper: download a file (for CSV/Excel export).
// Cookies are sent automatically (same-origin).
export async function downloadFile(url: string, filename: string) {
  const res = await fetch(url);
  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: 'Network error' }));
    throw new Error(err.error || res.statusText);
  }
  const blob = await res.blob();
  const blobUrl = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = blobUrl;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(blobUrl);
}

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { verifyPassword, generateToken, setAuthCookie } from '@/lib/auth';
import { isValidEmail, validatePassword, tooManyRequests } from '@/lib/validate';
import { rateLimitLogin } from '@/lib/rate-limit';

// P0.11/P0.12: rate limiting + email validation + password policy + timing-attack
// mitigation (bcrypt against dummy hash when user doesn't exist).
//
// P0.12 fix: parse req.json() ONCE — was calling it twice (once for rate-limit
// key, once for credentials), which threw on the second call → 500. Now we
// parse once and reuse.

// P0.13: DUMMY_HASH is a REAL bcrypt hash (cost 12, 60 chars).
// P0.12 used a 64-char string that wasn't a valid bcrypt hash →
// bcryptjs's early length check returned false in <1ms instead of
// doing the expensive bcrypt computation (~250ms). This made login
// with non-existent email MUCH faster than login with existing email
// + wrong password → timing-attack email enumeration was still possible.
//
// The fix: use a hash that bcryptjs will actually try to verify against,
// so the comparison takes ~250ms regardless of whether the user exists.
//
// The hash below was generated with `bcrypt.hashSync('dummy-password-for-timing-attack-mitigation-only', 12)`.
// It doesn't matter what password it hashes (no one can ever "match" it
// for a real login) — what matters is that bcrypt.compare takes the
// same time as a real hash.
const DUMMY_HASH = '$2b$12$xULoKaEdpcsZZTCcMRZgrOOnGj6UwIOGro1JRK9AD6WNabxLev7Yq';

export async function POST(req: NextRequest) {
  try {
    // P0.12: parse body ONCE, reuse everywhere.
    const body = await req.json().catch(() => ({}));
    const { email, password } = body;

    // P0.11: rate limit by IP+email to slow brute-force attacks
    // Use the (possibly empty) email from the request for the rate-limit key
    const rl = rateLimitLogin(req, typeof email === 'string' ? email : '');
    if (rl.limited) return tooManyRequests(rl.retryAfter, rl.message);

    // P0.11: validate input. Returns 401 (NOT 500) for missing/invalid creds.
    // Always return the SAME 401 message so attackers can't distinguish
    // "bad email" from "bad password".
    if (!isValidEmail(email) || typeof password !== 'string' || password.length === 0) {
      return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 });
    }

    const normalizedEmail = email.toLowerCase().trim();
    const user = await db.user.findUnique({ where: { email: normalizedEmail } });

    // P0.11: always run bcrypt against something, so timing doesn't reveal
    // whether the user exists. When user is null, use DUMMY_HASH.
    // bcrypt.compare against DUMMY_HASH will take ~250ms (same cost factor),
    // matching the timing of a real bcrypt comparison.
    const valid = await verifyPassword(password, user?.passwordHash ?? DUMMY_HASH);

    // P0.12: return 401 (NOT 500) for invalid credentials.
    // The earlier 500 was due to the double-req.json() bug.
    if (!user || !valid) {
      return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 });
    }

    const token = generateToken(user.id);
    // P0.12: do NOT return token in JSON body. Token is set ONLY in the
    // HttpOnly cookie. This closes the localStorage-XSS vector: an XSS
    // attack can't read the token because JS can't access HttpOnly cookies.
    // (Previous versions returned the token in JSON as a "Bearer fallback"
    // for preview proxies that strip Set-Cookie. That fallback is removed
    // — the preview proxy must be fixed to forward Set-Cookie instead.)
    const response = NextResponse.json({ data: { id: user.id, email: user.email, name: user.name } });
    return setAuthCookie(response, token);
  } catch (err: any) {
    // P0.11: don't leak err.message — generic 500
    console.error('Login error:', err);
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from 'next/server';
import {
  revokeToken,
  verifyTokenSignature,
  hashToken,
  maybeCleanupExpiredRevokedTokens,
  isProduction,
} from '@/lib/auth';

// POST /api/auth/logout
// Server-side token revocation: stores a SHA-256 HASH of the token in the
// RevokedToken table until its natural expiry, so even a stolen copy cannot
// be used after the legitimate user logs out.
//
// P0.13 fix: also delete the `__Host-auth-token` cookie (in production)
// AND the `auth-token` cookie (in dev). P0.12 changed the cookie name
// but logout was still trying to delete the old name → the cookie
// wasn't being cleared properly.
//
// P0.8 fixes:
//   - Returns 500 if revocation FAILS.
//   - Revokes BOTH Bearer token AND cookie token (if both present and different).
//   - revokeToken() throws on DB error (does not swallow).
//
// P0.9 fix:
//   - Logs the token HASH (not the token substring).
//
// Cleanup runs every 10 logouts OR if 1+ hour since last cleanup.
export async function POST(req: NextRequest) {
  const bearerToken = extractBearerToken(req);
  const cookieToken = extractCookieToken(req);

  // Revoke all distinct tokens present in the request.
  const tokensToRevoke = new Set<string>();
  if (bearerToken) tokensToRevoke.add(bearerToken);
  if (cookieToken) tokensToRevoke.add(cookieToken);

  let revocationFailed = false;
  const revokedHashes: string[] = [];

  for (const token of tokensToRevoke) {
    const verified = verifyTokenSignature(token);
    if (!verified) continue;
    const tokenHash = hashToken(token);
    try {
      await revokeToken(token, verified.userId);
      revokedHashes.push(tokenHash.substring(0, 12));
    } catch (e: any) {
      console.error('revokeToken failed for hash', tokenHash.substring(0, 12) + '...:', e?.message);
      revocationFailed = true;
    }
  }

  maybeCleanupExpiredRevokedTokens().catch(() => {});
  if (revokedHashes.length > 0) {
    console.log('logout: revoked', revokedHashes.length, 'token(s) with hashes:', revokedHashes);
  }

  // P0.13/P0.14: clear BOTH cookie names with proper attributes.
  // For __Host- cookies to be properly deleted, the deletion Set-Cookie
  // MUST include: Secure + Path=/ + no Domain (matches the __Host- prefix
  // requirements). NextResponse.cookies.delete() doesn't preserve these
  // attributes — it just sets the cookie to empty with Expires=1970.
  // So we use response.headers.append('Set-Cookie', ...) with the
  // full attribute string.
  const response = NextResponse.json(
    revocationFailed
      ? { success: false, error: 'Logout failed: token revocation failed. Stolen tokens may still be valid. Please try again.' }
      : { success: true },
    { status: revocationFailed ? 500 : 200 },
  );

  // Dev cookie: `auth-token` (no Secure needed for HTTP localhost)
  response.headers.append(
    'Set-Cookie',
    `auth-token=; HttpOnly; SameSite=Strict; Max-Age=0; Path=/; Expires=Thu, 01 Jan 1970 00:00:00 GMT`,
  );

  // Prod cookie: `__Host-auth-token` (requires Secure + Path=/ + no Domain)
  if (isProduction) {
    response.headers.append(
      'Set-Cookie',
      `__Host-auth-token=; HttpOnly; SameSite=Strict; Secure; Max-Age=0; Path=/; Expires=Thu, 01 Jan 1970 00:00:00 GMT`,
    );
  }
  return response;
}

function extractBearerToken(req: Request): string | null {
  const authHeader = req.headers.get('authorization');
  if (authHeader?.startsWith('Bearer ')) return authHeader.slice(7);
  return null;
}

function extractCookieToken(req: Request): string | null {
  const cookie = req.headers.get('cookie') || '';
  // P0.13: match either `auth-token` (dev) or `__Host-auth-token` (prod)
  const tokenMatch = cookie.match(/(?:__Host-)?auth-token=([^;]+)/);
  return tokenMatch ? tokenMatch[1] : null;
}

// frontend keeps the token and shows "service error, retry" UI.
// Only return 200 + null data when we KNOW the token is invalid
// (signature check failed, token is in blacklist, expired, etc.).
export async function GET(req: NextRequest) {
  let userId: string | null;
  try {
    userId = await getUserIdFromRequestAsync(req);
  } catch (e) {
    if (e instanceof DbUnavailableError) {
      // DB is down. Token may be valid; don't claim session is invalid.
      return NextResponse.json(
        { data: null, error: 'Service unavailable: database unreachable. Please retry.' },
        { status: 503 },
      );
    }
    // P0.10: unexpected error → 500, NOT 200. Frontend keeps token.
    console.error('Unexpected error in /api/auth/session (returning 500):', e);
    return NextResponse.json(
      { data: null, error: 'Internal server error. Please retry.' },
      { status: 500 },
    );
  }

  if (!userId) {
    // Token missing, invalid, expired, or revoked → real "session ended".
    // Frontend WILL clear the token on this response (200 + null data).
    return NextResponse.json({ data: null }, { status: 200 });
  }

  // Token is valid + not revoked. Look up the user record.
  let user;
  try {
    user = await db.user.findUnique({
      where: { id: userId },
      select: { id: true, email: true, name: true },
    });
  } catch (e) {
    // DB error fetching the user. Return 503 so frontend doesn't clear token.
    console.error('DB error fetching user in /api/auth/session (returning 503):', e);
    return NextResponse.json(
      { data: null, error: 'Service unavailable: database unreachable. Please retry.' },
      { status: 503 },
    );
  }

  return NextResponse.json({ data: user });
}

      if (!r.ok) return badRequest(r.error);

      db.category.findFirst({ where: { id: categoryId, userId } }),
    ]);

    if (!category) return NextResponse.json({ error: 'Category not found' }, { status: 404 });

    // P0.11: verify category type matches transaction type (catches mismatched imports)
    if (category.type !== type) {
      return badRequest(`Category type mismatch: category is ${category.type}, transaction is ${type}`);
    }

    const txCurrency: Currency = currency || account.currency;
    const amountInAccountCurrency = txCurrency === account.currency
      ? amt : convert(amt, txCurrency, account.currency);
    const delta = type === 'INCOME' ? amountInAccountCurrency : -amountInAccountCurrency;

    // Sanity check on converted amount
    if (!Number.isFinite(delta)) return badRequest('Currency conversion failed (invalid rate?)');

    // P0.12: snapshot the conversion for idempotent reversal later.
    // When this transaction is edited or deleted, we'll use sourceAmount
    // (the EXACT amount applied) instead of re-running convert() with

    const exchangeRateUsed = txCurrency === account.currency ? null : (amountInAccountCurrency / amt);

          type, amount: amt, currency: txCurrency,

          tags: tags ? String(tags).slice(0, 200) : null,

          destAmount: null,  // not a transfer
          exchangeRateUsed,
        },

    }

    if (!amtResult.ok) return badRequest(amtResult.error);

    if (currency !== undefined && !isCurrency(currency)) {

    ]);

    if (!destAccount) return NextResponse.json({ error: 'Destination account not found' }, { status: 404 });

    const txCurrency: Currency = currency || sourceAccount.currency;
    const sourceAmount = txCurrency === sourceAccount.currency
      ? amt : convert(amt, txCurrency, sourceAccount.currency);
    const destAmount = txCurrency === destAccount.currency
      ? amt : convert(amt, txCurrency, destAccount.currency);

    // P0.11: sanity check on converted amounts (NaN/Infinity can come from convert())
    if (!Number.isFinite(sourceAmount) || !Number.isFinite(destAmount)) {
      return badRequest('Currency conversion failed (invalid rate?)');
    }

    // P0.11: optional overdraft check (skip for LOAN accounts — they're meant to go negative)
    if (sourceAccount.type !== 'LOAN' && sourceAccount.balance < sourceAmount) {
      return NextResponse.json(
        { error: `Insufficient balance in source account (have ${sourceAccount.balance}, need ${sourceAmount})` },
        { status: 400 },
      );

    // P0.11: moved category lookup inside $transaction to close the race
    // (two concurrent transfers could both pass findFirst → both create → duplicate cats)

      });

          },
        });
      }

        where: { id: fromAccountId },
        data: { balance: { increment: -sourceAmount } },
      });
      await tx.account.update({
        where: { id: toAccountId },
        data: { balance: { increment: destAmount } },
      });

        data: {
          type: 'TRANSFER' as any,
          amount: amt,

          categoryId: transferCat.id,

          date: date ? new Date(date) : new Date(),
          note: note || `تحويل من ${sourceAccount.name} إلى ${destAccount.name}`,
          // P0.12: snapshot fields for idempotent reversal
          sourceAmount,
          destAmount,
          exchangeRateUsed: sourceAmount === amt ? null : (sourceAmount / amt),
        },
        include: { account: true, category: true, toAccount: true },
      });
    });

    return NextResponse.json({ data: transfer });
  } catch (err: any) {

import { NextRequest, NextResponse } from 'next/server';
import { db } from '@/lib/db';
import { EXCHANGE_RATES, type Currency } from '@/lib/currency';
import { requireAuthAsync } from '@/lib/auth';
import { handleAuthError } from '@/lib/api-errors';

// GET /api/stats?month=YYYY-MM&baseCurrency=EGP
export async function GET(req: NextRequest) {
  try {
    const userId = await requireAuthAsync(req);
    const url = new URL(req.url);
    const month = url.searchParams.get('month');
    const baseCurrency = (url.searchParams.get('baseCurrency') || 'EGP') as Currency;

    let start: Date, end: Date;
    if (month) {
      start = new Date(`${month}-01T00:00:00.000Z`);
      end = new Date(start);
      end.setMonth(end.getMonth() + 1);
    } else {
      start = new Date();
      start.setDate(1);
      start.setHours(0, 0, 0, 0);
      end = new Date(start);
      end.setMonth(end.getMonth() + 1);
    }

    const transactions = await db.transaction.findMany({
      where: { date: { gte: start, lt: end }, userId },
      include: { category: true, account: true },
    });

    const accounts = await db.account.findMany({ where: { userId } });

    const toBase = (amount: number, from: string) => {
      const fromCur = (from as Currency) || 'EGP';
      if (fromCur === baseCurrency) return amount;
      const inEgp = amount * EXCHANGE_RATES[fromCur];
      return inEgp / EXCHANGE_RATES[baseCurrency];
    };

    // Exclude transfers from BOTH expense and income
    const isTransfer = (t: any) =>
      t.type === 'TRANSFER' ||
      t.category?.nameAr === 'تحويلات' ||
      t.category?.nameEn === 'Transfers';

    const expenses = transactions.filter(t => t.type === 'EXPENSE' && !isTransfer(t));
    const incomes = transactions.filter(t => t.type === 'INCOME' && !isTransfer(t));

    const totalExpense = expenses.reduce((s, t) => s + toBase(t.amount, t.currency), 0);
    const totalIncome = incomes.reduce((s, t) => s + toBase(t.amount, t.currency), 0);

    // Total assets: exclude LOAN type
    const totalAssets = accounts
      .filter(a => a.type !== 'LOAN')
      .reduce((s, a) => s + toBase(a.balance, a.currency), 0);
    const totalLiabilities = accounts
      .filter(a => a.type === 'LOAN' && a.balance < 0)
      .reduce((s, a) => s + Math.abs(toBase(a.balance, a.currency)), 0);

    const byCategoryMap: Record<string, { total: number; count: number; category: any }> = {};
    expenses.forEach(t => {
      const cid = t.categoryId;
      if (!byCategoryMap[cid]) byCategoryMap[cid] = { total: 0, count: 0, category: t.category };
      byCategoryMap[cid].total += toBase(t.amount, t.currency);
      byCategoryMap[cid].count += 1;
    });

    // Daily chart: exclude transfers
    const daysInMonth = new Date(end.getFullYear(), end.getMonth(), 0).getDate();
    const byDay: { date: string; expense: number; income: number }[] = [];
    for (let day = 1; day <= daysInMonth; day++) {
      const dateStr = `${month || start.toISOString().slice(0, 7)}-${String(day).padStart(2, '0')}`;
      byDay.push({ date: dateStr, expense: 0, income: 0 });
    }
    transactions.forEach(t => {
      if (isTransfer(t)) return; // Exclude transfers from daily chart too
      const day = t.date.getDate();
      const idx = day - 1;
      if (idx >= 0 && idx < byDay.length) {
        if (t.type === 'EXPENSE') byDay[idx].expense += toBase(t.amount, t.currency);
        else if (t.type === 'INCOME') byDay[idx].income += toBase(t.amount, t.currency);
      }
    });

    return NextResponse.json({
      data: {
        totalExpense,
        totalIncome,
        balance: totalIncome - totalExpense,
        totalAssets,
        totalLiabilities,
        net: totalAssets - totalLiabilities,
        baseCurrency,
        expenseCount: expenses.length,
        incomeCount: incomes.length,
        byCategory: Object.values(byCategoryMap).sort((a: any, b: any) => b.total - a.total),
        byDay,
        month: month || start.toISOString().slice(0, 7),
      },
    });
  } catch (err: any) {
    return handleAuthError(err);
  }
}

const MAX_FILE_SIZE = 10 * 1024 * 1024; // 10MB

// Get keys case-insensitively
function findKey(row: Record<string, unknown>, patterns: string[]): string | undefined {
  const keys = Object.keys(row);
  for (const p of patterns) {
    const k = keys.find(k => k.toLowerCase().trim().includes(p.toLowerCase()));
    if (k) return k;
  }
  return undefined;
}

// P0.11: neutralize CSV injection — cells starting with =, +, -, @, tab, CR
function csvSafe(s: string): string {
  if (/^[=+\-@\t\r]/.test(s)) return `'${s}`;
  return s;
}

export async function POST(req: NextRequest) {
  try {
    const userId = await requireAuthAsync(req);
    const formData = await req.formData();
    const file = formData.get('file') as File;
    if (!file) return NextResponse.json({ error: 'No file uploaded' }, { status: 400 });

    // P0.11: file size limit (was: unbounded)
    if (file.size > MAX_FILE_SIZE) {
      return NextResponse.json({ error: `File too large (max ${MAX_FILE_SIZE / 1024 / 1024}MB)` }, { status: 400 });
    }

    const buf = await file.arrayBuffer();
    // P0.11: disable macro/formula evaluation to mitigate prototype-pollution gadgets
    const wb = XLSX.read(Buffer.from(buf), {
      type: 'buffer',
      cellDates: true,
      bookVBA: false,
      cellFormula: false,
      cellHTML: false,
    });
    const wsName = wb.SheetNames[0];
    if (!wsName) return NextResponse.json({ error: 'No sheets in workbook' }, { status: 400 });
    const ws = wb.Sheets[wsName];
    const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(ws, { defval: '' });

    if (rows.length === 0) return NextResponse.json({ error: 'No data found' }, { status: 400 });
    // P0.11: row count limit (was: unbounded)
    if (rows.length > MAX_ROWS) {
      return NextResponse.json({ error: `Too many rows (max ${MAX_ROWS})` }, { status: 400 });
    }

    const existingAccounts = await db.account.findMany({ where: { userId } });
    const existingCategories = await db.category.findMany({ where: { userId } });
    const accByArName = new Map(existingAccounts.map(a => [a.name, a]));
    const catByArName = new Map(existingCategories.map(c => [c.nameAr, c]));

    // P0.11: lookup existing transactions for duplicate detection (date+amount+accountId+note)
    const existingTxns = await db.transaction.findMany({
      where: { userId },
      select: { date: true, amount: true, accountId: true, note: true },
    });
    // Use a Set with composite key for O(1) lookup
    const existingKeys = new Set(
      existingTxns.map(t => `${t.date.getTime()}|${t.amount}|${t.accountId}|${t.note ?? ''}`),
    );

    // Fetch existing transfer category (we'll skip transfer rows entirely)
    const transferCat = await db.category.findFirst({
      where: { nameAr: 'تحويلات', userId, type: 'TRANSFER' as any },
    });

    let imported = 0, skipped = 0, duplicate = 0, transferSkipped = 0;

    for (const row of rows) {
      try {
        const dateKey = findKey(row, ['date', 'تاريخ']);
        const accountKey = findKey(row, ['account', 'حساب']);
        const catKey = findKey(row, ['category', 'فئة', 'تصنيف']);
        const typeKey = findKey(row, ['income/expense', 'نوع', 'type']);
        const amountKey = findKey(row, ['amount', 'مبلغ', 'value', 'sar']);
        const currencyKey = findKey(row, ['currency', 'عملة']);
        const noteKey = findKey(row, ['note', 'ملاحظة', 'description']);

        let date: Date = new Date();
        if (dateKey) {
          const dv = row[dateKey];
          if (dv instanceof Date) date = dv;
          else if (typeof dv === 'number') date = new Date((Math.floor(dv) - 25569) * 86400 * 1000);
          else if (typeof dv === 'string') {
            const str = dv.trim();
            if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(str)) {
              const [dd, mm, yyyy] = str.split('/');
              // P0.11: use Date.UTC instead of `new Date('YYYY-MM-DD')` (which is UTC midnight)
              date = new Date(Date.UTC(Number(yyyy), Number(mm) - 1, Number(dd)));
            } else date = new Date(str);
          }
        }
        if (isNaN(date.getTime())) { skipped++; continue; }

        // P0.11: validate amount via parseMoney (catches NaN/Infinity)
        const rawAmount = amountKey ? row[amountKey] : 0;
        const amtResult = parseMoney(rawAmount, { allowZero: false, allowNegative: true });
        if (!amtResult.ok) { skipped++; continue; }
        const amount = amtResult.value;

        const accName = accountKey ? String(row[accountKey] || '').trim() : '';
        if (!accName) { skipped++; continue; }

        // P0.11: validate currency BEFORE creating account
        const currencyRaw = (currencyKey ? String(row[currencyKey] || 'SAR') : 'SAR').toUpperCase();
        if (!isCurrency(currencyRaw)) { skipped++; continue; }
        const rowCurrency = currencyRaw as Currency;

        // Find or create account. P0.11: if account exists with different currency,
        // skip row (was: silently used account.currency and added raw amount)
        let account = accByArName.get(accName);
        if (!account) {
          account = await db.account.create({
            data: { name: accName, type: 'BANK', currency: rowCurrency, balance: 0, userId },
          });
          accByArName.set(accName, account);
        }
        // Currency mismatch check — don't silently corrupt
        if (account.currency !== rowCurrency) {
          // Convert the row amount to account currency, OR skip — choose skip for safety
          skipped++;
          continue;
        }

        // P0.11: detect Transfer rows and skip them (transfers can't round-trip via Excel
        // because we don't have a ToAccount column in the export format)
        const rowTypeStr = (typeKey ? String(row[typeKey] || 'Expense') : 'Expense').toLowerCase();
        if (rowTypeStr.startsWith('transfer')) {
          transferSkipped++;
          continue;
        }

        // P0.11: validate category type matches transaction type
        const catName = catKey ? String(row[catKey] || '').trim() : '';
        if (!catName || catName === 'None' || catName === '*') { skipped++; continue; }
        const isIncome = rowTypeStr === 'income' || amount < 0;
        const txType = isIncome ? 'INCOME' : 'EXPENSE';
        let category = catByArName.get(catName);
        if (!category) {
          category = await db.category.create({
            data: { nameAr: catName, nameEn: catName, type: txType as any, icon: 'tag', color: '#64748B', userId },
          });
          catByArName.set(catName, category);
        }
        // P0.11: if category exists but has wrong type for this row, skip (was: silently assigned)
        if (category.type !== txType && category.type !== 'TRANSFER' as any) {
          skipped++;
          continue;
        }

        const absAmount = Math.abs(amount);
        const note = noteKey ? String(row[noteKey] || '').slice(0, 500) : '';

        // P0.11: duplicate detection (date+amount+accountId+note)
        const dedupKey = `${date.getTime()}|${absAmount}|${account.id}|${note}`;
        if (existingKeys.has(dedupKey)) {
          duplicate++;
          continue;
        }
        existingKeys.add(dedupKey);

        // ATOMIC: update balance + create transaction
        // (balance delta is in account.currency because we validated currency match above)
        await db.$transaction(async (tx) => {
          const delta = txType === 'INCOME' ? absAmount : -absAmount;
          await tx.account.update({ where: { id: account!.id }, data: { balance: { increment: delta } } });
          await tx.transaction.create({
            data: {
              type: txType as any,
              amount: absAmount,
              currency: account!.currency,
              accountId: account!.id,
              categoryId: category!.id,
              date,
              note: note || null,
              userId,
            },
          });
        });

        imported++;
      } catch (e) {
        skipped++;
      }
    }

    return NextResponse.json({
      success: true,
      imported,
      skipped,
      duplicate,
      transferSkipped,
      message:
        `Imported ${imported} transaction(s). ` +
        `Skipped ${skipped} invalid row(s). ` +
        `Skipped ${duplicate} duplicate(s). ` +
        (transferSkipped > 0
          ? `Skipped ${transferSkipped} transfer(s) — transfers are not supported in Excel format (use Backup JSON to round-trip transfers). `
          : ''),
    });
  } catch (err: any) {
    return handleAuthError(err);
  }
}

  const [detailCatId, setDetailCatId] = useState<string | null>(null);

  const load = async () => {
    setLoading(true);
    try {
      const today = new Date().toISOString();
      const r = await api<{ data: TxItem[] }>(`/api/transactions?limit=9999&to=${today}`);
      setItems(r.data);
    } catch (e: any) { toast.error(e.message); }
    finally { setLoading(false); }
  };

  useEffect(() => { load(); }, []);

  // Date range filter
  const dateRange = useMemo(() => {
    const now = new Date();
    const end = now.toISOString().slice(0, 10);
    let start: string;
    switch (range) {
      case 'today': start = end; break;
      case 'week':
        const w = new Date(now); w.setDate(w.getDate() - 7);
        start = w.toISOString().slice(0, 10); break;
      case 'month':
        start = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-01`; break;
      case 'year':
        start = `${now.getFullYear()}-01-01`; break;
    }
    return { start, end };
  }, [range]);

  // Filter + group by category
  const categoryTotals = useMemo(() => {
    const map = new Map<string, { category: Category; total: number; count: number }>();
    items.forEach(tx => {
      if (tx.type === 'TRANSFER' || tx.category.nameAr === 'تحويلات' || tx.category.nameEn === 'Transfers') return; // Skip transfers
      const txDate = new Date(tx.date).toISOString().slice(0, 10);
      if (txDate < dateRange.start || txDate > dateRange.end) return;
      if (search) {
        const q = search.toLowerCase();
        const name = (lang === 'ar' ? tx.category.nameAr : tx.category.nameEn).toLowerCase();
        if (!name.includes(q)) return;
      }
      const cid = tx.category.id;
      if (!map.has(cid)) map.set(cid, { category: tx.category, total: 0, count: 0 });
      const entry = map.get(cid)!;
      entry.total += convert(tx.amount, tx.currency as Currency, baseCurrency);
      entry.count += 1;
    });
    return Array.from(map.values()).sort((a, b) => b.total - a.total);
  }, [items, dateRange, search, baseCurrency, lang]);

  const totalExpense = categoryTotals
    .filter(c => c.category.type === 'EXPENSE' || c.category.type === 'TRANSFER')
    .reduce((s, c) => s + c.total, 0);
  const totalIncome = categoryTotals
    .filter(c => c.category.type === 'INCOME')
    .reduce((s, c) => s + c.total, 0);

  const fmt = (n: number) => formatMoney(n, baseCurrency, lang);

  const rangeLabels: Record<RangeKey, string> = {
    today: lang === 'ar' ? 'اليوم' : 'Today',
    week: lang === 'ar' ? 'الأسبوع' : 'Week',
    month: lang === 'ar' ? 'الشهر' : 'Month',
    year: lang === 'ar' ? 'السنة' : 'Year',

        <p className="text-xs text-slate-500">{categoryTotals.length} {lang === 'ar' ? 'فئة' : 'categories'}</p>
      </header>

      {/* Range filter */}
      <div className="flex gap-1 bg-slate-100 dark:bg-slate-800 rounded-lg p-1">
        {(Object.keys(rangeLabels) as RangeKey[]).map(k => (
          <button
            key={k}
            onClick={() => setRange(k)}
            className={cn(
              'flex-1 py-1.5 rounded text-xs font-medium transition',
              range === k ? 'bg-white dark:bg-slate-900 text-blue-700 dark:text-blue-300 shadow-sm' : 'text-slate-500'
            )}
          >
            {rangeLabels[k]}
          </button>
        ))}
      </div>

      {/* Search */}
      <div className="relative">
        <Search className="absolute start-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
        <Input value={search} onChange={e => setSearch(e.target.value)} placeholder={lang === 'ar' ? 'بحث فئة...' : 'Search...'} className="ps-9" />
      </div>

      {/* Summary cards */}
      <div className="grid grid-cols-2 gap-2">
        <Card className="p-3 bg-emerald-50 dark:bg-emerald-950/30 border-emerald-200 dark:border-emerald-900">
          <div className="flex items-center gap-1.5 mb-0.5">
            <TrendingUp className="w-3.5 h-3.5 text-emerald-600" />
            <span className="text-[10px] font-semibold text-emerald-700 dark:text-emerald-400">{lang === 'ar' ? 'إجمالي الدخل' : 'Total Income'}</span>
          </div>
          <p className="text-sm font-bold text-emerald-700 dark:text-emerald-300 truncate">{fmt(totalIncome)}</p>
        </Card>
        <Card className="p-3 bg-red-50 dark:bg-red-950/30 border-red-200 dark:border-red-900">
          <div className="flex items-center gap-1.5 mb-0.5">
            <TrendingDown className="w-3.5 h-3.5 text-red-600" />
            <span className="text-[10px] font-semibold text-red-700 dark:text-red-400">{lang === 'ar' ? 'إجمالي المصروف' : 'Total Expense'}</span>
          </div>
          <p className="text-sm font-bold text-red-700 dark:text-red-300 truncate">{fmt(totalExpense)}</p>
        </Card>
      </div>

      {/* Category list */}
      {categoryTotals.length === 0 ? (
        <Card className="p-8 text-center">
          <Tag className="w-10 h-10 mx-auto text-slate-300 mb-2" />
          <p className="text-sm text-slate-400">{lang === 'ar' ? 'لا توجد فئات في هذه الفترة' : 'No categories in this period'}</p>
        </Card>
      ) : (
        <div className="space-y-2">
          {categoryTotals.map((c, i) => {
            const Icon = getIcon(c.category.icon);
            const isIncome = c.category.type === 'INCOME';
            const pct = totalExpense > 0 ? (c.total / totalExpense) * 100 : 0;
            return (
              <Card
                key={i}
                className="p-3 cursor-pointer hover:shadow-md transition"
                onClick={() => setDetailCatId(c.category.id)}
              >
                <div className="flex items-center gap-3">
                  <div className="w-10 h-10 rounded-full flex items-center justify-center text-white shrink-0" style={{ background: c.category.color }}>
                    <Icon className="w-5 h-5" />
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="flex justify-between items-baseline mb-1">
                      <span className="text-sm font-medium truncate">
                        {lang === 'ar' ? c.category.nameAr : c.category.nameEn}
                      </span>
                      <span className={cn('text-sm font-semibold shrink-0', isIncome ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400')}>
                        {isIncome ? '+' : '-'}{fmt(c.total)}
                      </span>
                    </div>
                    <div className="flex items-center gap-2">
                      <div className="h-1.5 bg-slate-100 dark:bg-slate-800 rounded-full overflow-hidden flex-1">
                        <div className="h-full rounded-full transition-all" style={{ width: `${Math.min(pct, 100)}%`, background: c.category.color }} />
                      </div>
                      <span className="text-[10px] text-slate-500 shrink-0">{c.count} {lang === 'ar' ? 'معاملة' : 'txns'}</span>

                  </div>
                </div>
              </Card>
            );
          })}
        </div>
      )}

      {/* Detail modal */}
      {detailCatId && (
        <CategoryDetailModal
          categoryId={detailCatId}
          dateRange={dateRange}
          onClose={() => setDetailCatId(null)}
        />
      )}
    </div>
  );
}

/** #4: Category Detail Modal - shows all transactions for a category in a given month */
function CategoryDetailModal({
  categoryId, month, onClose,
}: {
  categoryId: string;
  month: string;
  onClose: () => void;
}) {
  const { t, lang } = useT();
  const baseCurrency = useSettings(s => s.baseCurrency);
  const [txs, setTxs] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const start = `${month}-01`;
    const end = new Date(start);
    end.setMonth(end.getMonth() + 1);
    api<{ data: any[] }>(`/api/transactions?categoryId=${categoryId}&from=${start}&to=${end.toISOString()}&limit=500`)
      .then(r => setTxs(r.data))
      .finally(() => setLoading(false));
  }, [categoryId, month]);

  const fmt = (n: number, cur: any = baseCurrency) => formatMoney(n, cur, lang);
  const total = txs.reduce((s, t) => s + convert(t.amount, t.currency as Currency, baseCurrency), 0);

  return (
    <Dialog open={true} onOpenChange={(o) => !o && onClose()}>
      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>
            {txs[0] ? (lang === 'ar' ? txs[0].category.nameAr : txs[0].category.nameEn) : (lang === 'ar' ? 'تفاصيل الفئة' : 'Category Details')}
          </DialogTitle>
        </DialogHeader>
        {loading ? (
          <Skeleton className="h-40 w-full" />
        ) : (
          <>
            <Card className="p-4 bg-blue-600 text-white border-0">
              <p className="text-xs text-blue-100">{lang === 'ar' ? 'إجمالي الإنفاق' : 'Total Spending'}</p>
              <p className="text-2xl font-bold">{fmt(total)}</p>
              <p className="text-xs text-blue-200 mt-1">{txs.length} {lang === 'ar' ? 'معاملة' : 'transactions'}</p>
            </Card>
            <div className="mt-3 space-y-1.5 max-h-72 overflow-y-auto">
              {txs.map(tx => {
                const Icon = getIcon(tx.category?.icon || 'tag');
                const isIncome = tx.type === 'INCOME';
                return (
                  <div key={tx.id} className="flex items-center gap-2 p-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 rounded-lg">
                    <div className="w-8 h-8 rounded-full flex items-center justify-center text-white shrink-0" style={{ background: tx.category?.color || '#64748B' }}>
                      <Icon className="w-4 h-4" />
                    </div>
                    <div className="flex-1 min-w-0">
                      <p className="text-sm font-medium truncate">{tx.account?.name}</p>
                      <p className="text-[10px] text-slate-400 truncate">
                        {new Date(tx.date).toLocaleDateString(lang === 'ar' ? 'ar-EG' : 'en-US', { day: 'numeric', month: 'short' })}
                        {tx.note ? ` • ${tx.note}` : ''}
                      </p>
                    </div>
                    <span className={cn('text-xs font-semibold shrink-0', isIncome ? 'text-emerald-600' : 'text-red-600')}>
                      {isIncome ? '+' : '-'}{fmt(tx.amount, tx.currency)}
                    </span>
                  </div>
                );
              })}
            </div>
          </>
        )}
      </DialogContent>
    </Dialog>
  );
}

            <p className="text-xs text-slate-500 mt-0.5">
              {lang === 'ar'
                ? 'حمّل ملف APK وثبّته على موبايلك بدون نشر على Google Play'
                : 'Download APK file and install on your phone without Google Play'}
            </p>
          </div>
        </div>
        <div className="text-xs text-slate-500 mb-3 space-y-1">
          <div className="flex items-center justify-between">
            <span>{lang === 'ar' ? 'الإصدار' : 'Version'}:</span>
            <span className="font-mono">1.0.0</span>
          </div>
          <div className="flex items-center justify-between">
            <span>{lang === 'ar' ? 'الحجم' : 'Size'}:</span>
            <span className="font-mono">5.5 MB</span>
          </div>
          <div className="flex items-center justify-between">
            <span>{lang === 'ar' ? 'يدعم' : 'Supports'}:</span>
            <span className="font-mono">Android 7.0+</span>
          </div>
        </div>
        <a
          href="/MyWallet.apk"
          download="MyWallet.apk"
          className="flex items-center justify-center gap-2 w-full h-12 rounded-xl bg-blue-600 hover:bg-blue-700 active:scale-95 transition text-white font-semibold text-base shadow-md"
        >
          <Download className="w-5 h-5" />
          {lang === 'ar' ? 'حمّل ملف APK الآن' : 'Download APK Now'}
        </a>
        <p className="text-[10px] text-slate-400 mt-2 text-center">
          {lang === 'ar'
            ? 'بعد التحميل: افتح الـ File Manager واضغط على الملف لتثبيته'
            : 'After download: open File Manager and tap the file to install'}
        </p>
      </Card>

      {/* Categories Management */}
      <CategoriesManager />

      <DataTools />

          {t('resetData')}
        </Button>
      </Card>

      {/* About */}
      <Card className="p-4">
        <div className="flex items-center gap-3 mb-3">
          <Info className="w-5 h-5 text-slate-500" />
          <h2 className="font-semibold">{t('about')}</h2>
        </div>

          <p>Next.js 16 + PWA + Prisma</p>
        </div>
      </Card>

      {/* Logout */}
      {onLogout && (
        <Card className="p-4">
          <Button
            variant="outline"
            onClick={async () => {
              // Clear stored token client-side + call logout API to clear cookie
              setStoredToken(null);
              await fetch('/api/auth/logout', { method: 'POST' });
              onLogout();
            }}
            className="w-full h-12 text-red-600 border-red-200 hover:bg-red-50"
          >
            {lang === 'ar' ? 'تسجيل الخروج' : 'Logout'}
          </Button>
        </Card>
      )}

      <div className="h-4" />
    </div>
  );
}
