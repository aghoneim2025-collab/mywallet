   336→   336→function AccountDetailModal({
   337→   337→  accountId,
   338→   338→  onClose,
   339→   339→  onEditTx,
   340→   340→}: {
   341→   341→  accountId: string;
   342→   342→  onClose: () => void;

   344→   344→}) {
   345→   345→  const { t, lang } = useT();
   346→   346→  const baseCurrency = useSettings(s => s.baseCurrency);
   347→   347→  const [account, setAccount] = useState<Account | null>(null);

   349→   349→  const [loading, setLoading] = useState(true);
   350→   350→
   351→   351→  useEffect(() => {
   352→   352→    Promise.all([
   353→   353→      api<{ data: Account[] }>('/api/accounts'),
   354→   354→      api<{ data: any[] }>(`/api/transactions?accountId=${accountId}&limit=200`),
   355→   355→    ]).then(([accs, txs]) => {
   356→   356→      const acc = accs.data.find(a => a.id === accountId) || null;
   357→   357→      setAccount(acc);
   358→   358→      setTransactions(txs.data);
   359→   359→    }).finally(() => setLoading(false));
   360→   360→  }, [accountId]);
   361→   361→
   362→   362→  const fmt = (n: number, cur: any = baseCurrency) => formatMoney(n, cur, lang);
   363→   363→  const convertToBase = (balance: number, fromCurrency: string) =>
   364→   364→    convert(balance, fromCurrency as Currency, baseCurrency);

   368→   368→    .filter(t => t.type === 'INCOME')

   372→   372→    .reduce((s, t) => s + convertToBase(t.amount, t.currency), 0);
   373→   373→  const totalTransfers = transactions
   374→   374→    .filter(t => t.type === 'TRANSFER')
   375→   375→    .reduce((s, t) => s + convertToBase(t.amount, t.currency), 0);

   380→   380→        <DialogHeader>

   384→   384→                <div
   385→   385→                  className="w-8 h-8 rounded-lg flex items-center justify-center text-white"
   386→   386→                  style={{ background: account.color }}
   387→   387→                >
   388→   388→                  {(() => {
   389→   389→                    const I = getIcon(account.icon);
   390→   390→                    return <I className="w-4 h-4" />;
   391→   391→                  })()}

   397→   397→        </DialogHeader>
