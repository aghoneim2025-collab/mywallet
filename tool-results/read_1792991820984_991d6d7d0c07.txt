   465→              {/* Legend on the right */}
   466→              <div className="flex-1 space-y-1.5">
   467→                {pieData.map((entry, i) => {
   468→                  const pct = stat.totalExpense > 0 ? (entry.value / stat.totalExpense) * 100 : 0;
   469→                  return (
   470→                    <div key={i} className="flex items-center gap-2">
   471→                      <div
   472→                        className="w-2.5 h-2.5 rounded-full shrink-0"
   473→                        style={{ background: entry.color }}
   474→                      />
   475→                      <span className="text-[11px] font-medium flex-1 truncate">{entry.name}</span>
   476→                      <span className="text-[10px] text-slate-500">{pct.toFixed(1)}%</span>
   477→                      <span className="text-[11px] font-semibold">{fmt(entry.value)}</span>
   478→                    </div>
   479→                  );
   480→                })}
   481→              </div>
   482→            </div>
   483→          </Card>
   484→        )}
   485→      </section>
   486→
   487→      {/* Top categories list */}
   488→      <section>
   489→        <h2 className="text-sm font-semibold mb-2">{t('topCategories')}</h2>
   490→        {sortedCategories.length === 0 ? (
   491→          <Card className="p-6 text-center">
   492→            <p className="text-sm text-slate-400">{t('noTransactionsYet')}</p>
   493→          </Card>
   494→        ) : (
   495→          <div className="space-y-2">
   496→            {sortedCategories.slice(0, 10).map((c, i) => {
   497→              const Icon = getIcon(c.category.icon);
   498→              const pct = stat.totalExpense > 0 ? (c.total / stat.totalExpense) * 100 : 0;
   499→              return (
   500→                <Card key={i} className="p-3 cursor-pointer hover:shadow-md transition" onClick={() => setDetailCategoryId(c.category.id)}>
   501→                  <div className="flex items-center gap-3">
   502→                    <div
   503→                      className="w-9 h-9 rounded-full flex items-center justify-center text-white shrink-0"
   504→                      style={{ background: c.category.color }}
   505→                    >
   506→                      <Icon className="w-4 h-4" />
   507→                    </div>
   508→                    <div className="flex-1 min-w-0">
   509→                      <div className="flex justify-between items-baseline mb-1">
   510→                        <span className="text-sm font-medium truncate">
   511→                          {lang === 'ar' ? c.category.nameAr : c.category.nameEn}
   512→                        </span>
   513→                        <span className="text-sm font-semibold shrink-0">{fmt(c.total)}</span>
   514→                      </div>
   515→                      <div className="flex items-center gap-2">
   516→                        <div className="h-1.5 bg-slate-100 dark:bg-slate-800 rounded-full overflow-hidden flex-1">
   517→                          <div
   518→                            className="h-full rounded-full transition-all"
   519→                            style={{ width: `${pct}%`, background: c.category.color }}
   520→                          />
   521→                        </div>
   522→                        <span className="text-[10px] text-slate-500 shrink-0">{pct.toFixed(1)}%</span>
   523→                      </div>
   524→                    </div>
   525→                  </div>
   526→                </Card>
   527→              );
   528→            })}
   529→          </div>
   530→        )}
   531→      </section>
   532→
   533→      {/* Export */}
   534→      <section className="grid grid-cols-2 gap-2">
   535→        <Button
   536→          onClick={handleExportCSV}
   537→          variant="outline"
   538→          className="h-12 rounded-xl bg-emerald-50 dark:bg-emerald-950/30 border-emerald-200 dark:border-emerald-900 text-emerald-700 dark:text-emerald-300 hover:bg-emerald-100"
   539→        >
   540→          <Download className="w-5 h-5" /> Excel
   541→        </Button>
   542→        <Button
   543→          onClick={handleExportPDF}
   544→          variant="outline"
   545→          className="h-12 rounded-xl bg-red-50 dark:bg-red-950/30 border-red-200 dark:border-red-900 text-red-700 dark:text-red-300 hover:bg-red-100"
   546→        >
   547→          <FileText className="w-5 h-5" /> PDF
   548→        </Button>
   549→      </section>
   550→
   551→      {/* Category Detail Modal */}
   552→      {detailCategoryId && (
   553→        <CategoryDetailModal
   554→          categoryId={detailCategoryId}
   555→          month={month}
   556→          onClose={() => setDetailCategoryId(null)}
   557→        />
   558→      )}
   559→    </div>
   560→  );
   561→}

   563→/** #4: Category Detail Modal - shows all transactions for a category in a given month */
   564→function CategoryDetailModal({
   565→  categoryId, month, onClose,
   566→}: {
   567→  categoryId: string;
   568→  month: string;
   569→  onClose: () => void;
   570→}) {
   571→  const { t, lang } = useT();
   572→  const baseCurrency = useSettings(s => s.baseCurrency);
   573→  const [txs, setTxs] = useState<any[]>([]);
   574→  const [loading, setLoading] = useState(true);
   575→
   576→  useEffect(() => {
   577→    const start = `${month}-01`;
   578→    const end = new Date(start);
   579→    end.setMonth(end.getMonth() + 1);
   580→    api<{ data: any[] }>(`/api/transactions?categoryId=${categoryId}&from=${start}&to=${end.toISOString()}&limit=500`)
   581→      .then(r => setTxs(r.data))
   582→      .finally(() => setLoading(false));
   583→  }, [categoryId, month]);
   584→
   585→  const fmt = (n: number, cur: any = baseCurrency) => formatMoney(n, cur, lang);
   586→  const total = txs.reduce((s, t) => s + convert(t.amount, t.currency as Currency, baseCurrency), 0);
   587→
   588→  return (
   589→    <Dialog open={true} onOpenChange={(o) => !o && onClose()}>
   590→      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   591→        <DialogHeader>
   592→          <DialogTitle>
   593→            {txs[0] ? (lang === 'ar' ? txs[0].category.nameAr : txs[0].category.nameEn) : (lang === 'ar' ? 'تفاصيل الفئة' : 'Category Details')}
   594→          </DialogTitle>
   595→        </DialogHeader>
   596→        {loading ? (
   597→          <Skeleton className="h-40 w-full" />
   598→        ) : (
   599→          <>
   600→            <Card className="p-4 bg-blue-600 text-white border-0">
   601→              <p className="text-xs text-blue-100">{lang === 'ar' ? 'إجمالي الإنفاق' : 'Total Spending'}</p>
   602→              <p className="text-2xl font-bold">{fmt(total)}</p>
   603→              <p className="text-xs text-blue-200 mt-1">{txs.length} {lang === 'ar' ? 'معاملة' : 'transactions'}</p>
   604→            </Card>
   605→            <div className="mt-3 space-y-1.5 max-h-72 overflow-y-auto">
   606→              {txs.map(tx => {
   607→                const Icon = getIcon(tx.category?.icon || 'tag');
   608→                const isIncome = tx.type === 'INCOME';
   609→                return (
   610→                  <div key={tx.id} className="flex items-center gap-2 p-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 rounded-lg">
   611→                    <div className="w-8 h-8 rounded-full flex items-center justify-center text-white shrink-0" style={{ background: tx.category?.color || '#64748B' }}>
   612→                      <Icon className="w-4 h-4" />
   613→                    </div>
   614→                    <div className="flex-1 min-w-0">
   615→                      <p className="text-sm font-medium truncate">{tx.account?.name}</p>
   616→                      <p className="text-[10px] text-slate-400 truncate">
   617→                        {new Date(tx.date).toLocaleDateString(lang === 'ar' ? 'ar-EG' : 'en-US', { day: 'numeric', month: 'short' })}
   618→                        {tx.note ? ` • ${tx.note}` : ''}
   619→                      </p>
   620→                    </div>
   621→                    <span className={cn('text-xs font-semibold shrink-0', isIncome ? 'text-emerald-600' : 'text-red-600')}>
   622→                      {isIncome ? '+' : '-'}{fmt(tx.amount, tx.currency)}
   623→                    </span>
   624→                  </div>
   625→                );
   626→              })}
   627→            </div>
   628→          </>
   629→        )}
   630→      </DialogContent>
   631→    </Dialog>
   632→  );
   633→}
