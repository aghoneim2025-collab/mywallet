    96→    if (!search) return true;

    98→    return a.name.toLowerCase().includes(q) ||
    99→      a.currency.toLowerCase().includes(q) ||
   100→      a.type.toLowerCase().includes(q);
   101→  });
   102→
   103→  // Group accounts
   104→  const groupedAccounts = ACCOUNT_GROUPS.map(group => ({
   105→    ...group,
   106→    accounts: filteredAccounts.filter(a => a.type === group.type),
   107→  })).filter(g => g.accounts.length > 0);
   108→

   110→  const totalAssets = accounts
   111→    .filter(a => !['LOAN'].includes(a.type))

   116→  const net = totalAssets - totalLiabilities;

   132→      toast.error(e.message);
   133→    }

   137→    if (!confirm(lang === 'ar' ? 'حذف الحساب؟' : 'Delete account?')) return;

   139→      await api(`/api/accounts/${id}`, { method: 'DELETE' });
   140→      toast.success(lang === 'ar' ? 'تم الحذف' : 'Deleted');
   141→      load();
   142→    } catch (e: any) {
   143→      toast.error(e.message);
   144→    }
   145→  };
   146→
   147→  if (loading) {
   148→    return (
   149→      <div className="p-4 space-y-3">
   150→        <Skeleton className="h-32 w-full rounded-2xl" />

   153→    );
   154→  }
   155→
   156→  return (
   157→    <div className="p-4 space-y-5 pb-4">
   158→      <header className="flex justify-between items-center">
   159→        <div>
   160→          <h1 className="text-2xl font-bold">{t('accounts')}</h1>
   161→          <p className="text-xs text-slate-500">{accounts.length} {t('accounts')}</p>
   162→        </div>
   163→        <Button onClick={() => setModal({ open: true })} size="sm" className="rounded-full bg-blue-600 hover:bg-blue-700">
   164→          <Plus className="w-4 h-4" /> {t('add')}
   165→        </Button>
   166→      </header>
   167→
   168→      {/* Search box */}
   169→      <div className="relative">
   170→        <Search className="absolute start-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   171→        <Input
   172→          value={search}
   173→          onChange={e => setSearch(e.target.value)}
   174→          placeholder={lang === 'ar' ? 'بحث بالاسم أو العملة...' : 'Search by name or currency...'}
   175→          className="ps-9"
   176→        />
   177→      </div>
   178→
   179→      {/* Summary Card: Assets / Liabilities / Net */}
   180→      <div className="grid grid-cols-3 gap-2">

   183→            <TrendingUp className="w-3.5 h-3.5 text-emerald-600" />
   184→            <span className="text-[10px] font-semibold text-emerald-700 dark:text-emerald-400">
   185→              {lang === 'ar' ? 'الأصول' : 'Assets'}
   186→            </span>
   187→          </div>
   188→          <p className="text-sm font-bold text-emerald-700 dark:text-emerald-300 truncate">{fmt(totalAssets)}</p>
   189→        </Card>
   190→        <Card className="p-3 bg-red-50 dark:bg-red-950/30 border-red-200 dark:border-red-900">
   191→          <div className="flex items-center gap-1.5 mb-1">
   192→            <TrendingDown className="w-3.5 h-3.5 text-red-600" />
   193→            <span className="text-[10px] font-semibold text-red-700 dark:text-red-400">
   194→              {lang === 'ar' ? 'الخصوم' : 'Liabilities'}
   195→            </span>
   196→          </div>
   197→          <p className="text-sm font-bold text-red-700 dark:text-red-300 truncate">{fmt(totalLiabilities)}</p>
   198→        </Card>
   199→        <Card className="p-3 bg-blue-50 dark:bg-blue-950/30 border-blue-200 dark:border-blue-900">
   200→          <div className="flex items-center gap-1.5 mb-1">
   201→            <Wallet className="w-3.5 h-3.5 text-blue-600" />
   202→            <span className="text-[10px] font-semibold text-blue-700 dark:text-blue-400">
   203→              {lang === 'ar' ? 'الصافي' : 'Net'}

   209→        </Card>
   210→      </div>
   211→
   212→      {/* Grouped accounts */}
   213→      {groupedAccounts.map(group => {
   214→        const groupTotal = group.accounts.reduce(
   215→          (s, a) => s + convertToBase(a.balance, a.currency),

   224→                <div className={cn(
   225→                  "w-7 h-7 rounded-lg flex items-center justify-center text-white",
   226→                  group.isAsset ? "bg-blue-600" : "bg-red-600"
   227→                )}>
   228→                  <GroupIcon className="w-4 h-4" />
   229→                </div>
   230→                <h3 className="text-sm font-semibold">{lang === 'ar' ? group.arLabel : group.enLabel}</h3>
   231→                <span className="text-xs text-slate-400">({group.accounts.length})</span>
   232→              </div>
   233→              <span className={cn(
   234→                "text-xs font-semibold",

   256→                        <Icon className="w-5 h-5" />

   301→          </div>

   304→
   305→      {/* Empty state */}

   308→          <Wallet className="w-12 h-12 mx-auto text-slate-300 mb-2" />

   312→        </Card>

   314→
   315→      <AccountModal
   316→        key={modal.data?.id || 'new'}

   325→        <AccountDetailModal
   326→          accountId={detailAccountId}
   327→          onClose={() => setDetailAccountId(null)}
   328→          onEditTx={() => {/* TODO: implement edit from detail */}}

   335→/** Account detail view - shows account info + recent transactions */
   336→function AccountDetailModal({
   337→  accountId,
   338→  onClose,
   339→  onEditTx,
   340→}: {
   341→  accountId: string;
   342→  onClose: () => void;
   343→  onEditTx?: (tx: any) => void;
   344→}) {
   345→  const { t, lang } = useT();
   346→  const baseCurrency = useSettings(s => s.baseCurrency);
   347→  const [account, setAccount] = useState<Account | null>(null);
   348→  const [transactions, setTransactions] = useState<any[]>([]);
   349→  const [loading, setLoading] = useState(true);
   350→
   351→  useEffect(() => {
   352→    Promise.all([
   353→      api<{ data: Account[] }>('/api/accounts'),
   354→      api<{ data: any[] }>(`/api/transactions?accountId=${accountId}&limit=200`),
   355→    ]).then(([accs, txs]) => {
   356→      const acc = accs.data.find(a => a.id === accountId) || null;
   357→      setAccount(acc);
   358→      setTransactions(txs.data);
   359→    }).finally(() => setLoading(false));
   360→  }, [accountId]);
   361→
   362→  const fmt = (n: number, cur: any = baseCurrency) => formatMoney(n, cur, lang);
   363→  const convertToBase = (balance: number, fromCurrency: string) =>
   364→    convert(balance, fromCurrency as Currency, baseCurrency);
   365→
   366→  // Calculate totals for this account
   367→  const totalIncome = transactions
   368→    .filter(t => t.type === 'INCOME')
   369→    .reduce((s, t) => s + convertToBase(t.amount, t.currency), 0);
   370→  const totalExpense = transactions
   371→    .filter(t => t.type === 'EXPENSE')
   372→    .reduce((s, t) => s + convertToBase(t.amount, t.currency), 0);
   373→  const totalTransfers = transactions
   374→    .filter(t => t.type === 'TRANSFER')
   375→    .reduce((s, t) => s + convertToBase(t.amount, t.currency), 0);
   376→
   377→  return (
   378→    <Dialog open={true} onOpenChange={(o) => !o && onClose()}>
   379→      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   380→        <DialogHeader>
   381→          <DialogTitle className="flex items-center gap-2">
   382→            {account && (
   383→              <>
   384→                <div
   385→                  className="w-8 h-8 rounded-lg flex items-center justify-center text-white"
   386→                  style={{ background: account.color }}
   387→                >
   388→                  {(() => {
   389→                    const I = getIcon(account.icon);
   390→                    return <I className="w-4 h-4" />;
   391→                  })()}
   392→                </div>
   393→                {account.name}
   394→              </>
   395→            )}
   396→          </DialogTitle>
   397→        </DialogHeader>
   398→
   399→        {loading ? (
   400→          <div className="py-8 text-center">
   401→            <Skeleton className="h-32 w-full" />
   402→          </div>
   403→        ) : account ? (
   404→          <>
   405→            {/* Account balance */}
   406→            <Card className="p-5 bg-gradient-to-br from-blue-600 to-blue-700 text-white border-0">
   407→              <p className="text-xs text-blue-100">{lang === 'ar' ? 'الرصيد الحالي' : 'Current Balance'}</p>
   408→              <p className="text-3xl font-bold mt-1">
   409→                {formatMoney(account.balance, account.currency as Currency, lang)}
   410→              </p>
   411→              {account.currency !== baseCurrency && (
   412→                <p className="text-xs text-blue-200 mt-1">
   413→                  ≈ {fmt(convertToBase(account.balance, account.currency))}
   414→                </p>
   415→              )}
   416→            </Card>
   417→
   418→            {/* Mini summary */}
   419→            <div className="grid grid-cols-3 gap-2 mt-2">
   420→              <div className="p-2 rounded-lg bg-emerald-50 dark:bg-emerald-950/30 text-center">
   421→                <p className="text-[10px] text-emerald-700 dark:text-emerald-400">{lang === 'ar' ? 'دخل' : 'Income'}</p>
   422→                <p className="text-xs font-bold text-emerald-700 dark:text-emerald-300 truncate">{fmt(totalIncome)}</p>
   423→              </div>
   424→              <div className="p-2 rounded-lg bg-red-50 dark:bg-red-950/30 text-center">
   425→                <p className="text-[10px] text-red-700 dark:text-red-400">{lang === 'ar' ? 'مصروف' : 'Expense'}</p>
   426→                <p className="text-xs font-bold text-red-700 dark:text-red-300 truncate">{fmt(totalExpense)}</p>
   427→              </div>
   428→              <div className="p-2 rounded-lg bg-slate-50 dark:bg-slate-800/50 text-center">
   429→                <p className="text-[10px] text-slate-700 dark:text-slate-300">{lang === 'ar' ? 'تحويل' : 'Transfer'}</p>
   430→                <p className="text-xs font-bold text-slate-700 dark:text-slate-300 truncate">{fmt(totalTransfers)}</p>
   431→              </div>
   432→            </div>
   433→
   434→            {/* Transactions list */}
   435→            <div className="mt-4">
   436→              <h3 className="text-sm font-semibold mb-2">
   437→                {transactions.length} {lang === 'ar' ? 'معاملة' : 'transactions'}
   438→              </h3>
   439→              <div className="max-h-72 overflow-y-auto -mx-2">
   440→                {transactions.length === 0 ? (
   441→                  <p className="text-center text-sm text-slate-400 py-4">{t('noTransactionsYet')}</p>
   442→                ) : (
   443→                  transactions.slice(0, 50).map(tx => {
   444→                    const Icon = getIcon(tx.category?.icon || 'tag');
   445→                    const isIncome = tx.type === 'INCOME';
   446→                    const isTransfer = tx.type === 'TRANSFER';
   447→                    return (
   448→                      <div key={tx.id} className="flex items-center gap-3 p-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 rounded-lg transition">
   449→                        <div
   450→                          className="w-8 h-8 rounded-full flex items-center justify-center text-white shrink-0"
   451→                          style={{ background: isTransfer ? '#64748B' : tx.category?.color || '#64748B' }}
   452→                        >
   453→                          <Icon className="w-4 h-4" />
   454→                        </div>
   455→                        <div className="flex-1 min-w-0">
   456→                          <p className="text-sm font-medium truncate">
   457→                            {isTransfer
   458→                              ? (lang === 'ar' ? 'تحويل' : 'Transfer')
   459→                              : (lang === 'ar' ? tx.category?.nameAr : tx.category?.nameEn)}
   460→                          </p>
   461→                          <p className="text-[10px] text-slate-400 truncate">
   462→                            {new Date(tx.date).toLocaleDateString(lang === 'ar' ? 'ar-EG' : 'en-US', { day: 'numeric', month: 'short', year: 'numeric' })}
   463→                            {tx.note ? ` • ${tx.note}` : ''}
   464→                          </p>
   465→                        </div>
   466→                        <span className={cn(
   467→                          'text-xs font-semibold shrink-0',
   468→                          isIncome
   469→                            ? 'text-emerald-600 dark:text-emerald-400'
   470→                            : isTransfer
   471→                              ? 'text-slate-600 dark:text-slate-300'
   472→                              : 'text-red-600 dark:text-red-400'
   473→                        )}>
   474→                          {isIncome ? '+' : isTransfer ? '⇄ ' : '-'}{fmt(tx.amount, tx.currency)}
   475→                        </span>
   476→                      </div>
   477→                    );
   478→                  })
   479→                )}
   480→              </div>
   481→            </div>
   482→          </>
   483→        ) : (
   484→          <p className="text-center text-sm text-slate-400 py-4">{lang === 'ar' ? 'الحساب غير موجود' : 'Account not found'}</p>
   485→        )}
   486→      </DialogContent>
   487→    </Dialog>
   488→  );
   489→}

   491→function AccountModal({
   492→  open, onOpenChange, initial, onSave,
   493→}: {
   494→  open: boolean;
   495→  onOpenChange: (o: boolean) => void;
   496→  initial?: Account | null;
   497→  onSave: (data: any) => void;
   498→}) {
   499→  const { t, lang } = useT();
   500→  const [name, setName] = useState(initial?.name || '');
   501→  const [type, setType] = useState(initial?.type || 'CASH');
   502→  const [currency, setCurrency] = useState<Currency>(initial?.currency as Currency || 'EGP');
   503→  const [balance, setBalance] = useState(initial ? initial.balance.toString() : '');
   504→  const [color, setColor] = useState(initial?.color || COLORS[0]);
   505→  const [icon, setIcon] = useState(initial?.icon || 'wallet');
   506→  const [isDefault, setIsDefault] = useState(initial?.isDefault || false);
   507→
   508→  const submit = () => {
   509→    if (!name || !type || !currency) {
   510→      toast.error(lang === 'ar' ? 'أدخل الاسم والنوع والعملة' : 'Enter name, type, currency');
   511→      return;
   512→    }
   513→    onSave({
   514→      id: initial?.id,
   515→      name, type, currency,
   516→      balance: parseFloat(balance) || 0,
   517→      color, icon, isDefault,
   518→    });
   519→  };
   520→
   521→  return (
   522→    <Dialog open={open} onOpenChange={onOpenChange}>
   523→      <DialogContent>
   524→        <DialogHeader>
   525→          <DialogTitle>{initial ? t('editAccount') : t('addAccount')}</DialogTitle>
   526→        </DialogHeader>
   527→        <div className="space-y-3 max-h-[70vh] overflow-y-auto">
   528→          <div className="flex justify-center py-2">
   529→            <div
   530→              className="w-16 h-16 rounded-2xl flex items-center justify-center text-white"

   532→            >
   533→              <DynamicIcon name={icon} className="w-7 h-7" />

   535→          </div>
   536→
   537→          <div className="space-y-1.5">
   538→            <Label>{t('account')}</Label>
   539→            <Input value={name} onChange={e => setName(e.target.value)} placeholder="..." />
   540→          </div>
   541→
   542→          <div className="grid grid-cols-2 gap-2">
   543→            <div className="space-y-1.5">
   544→              <Label>{t('type')}</Label>
   545→              <Select value={type} onValueChange={setType}>
   546→                <SelectTrigger><SelectValue /></SelectTrigger>
   547→                <SelectContent>
   548→                  {TYPES.map(t => <SelectItem key={t.value} value={t.value}>{lang === 'ar' ? t.labelAr : t.labelEn}</SelectItem>)}
   549→                </SelectContent>
   550→              </Select>
   551→            </div>
   552→            <div className="space-y-1.5">
   553→              <Label>{t('currency')}</Label>
   554→              <Select value={currency} onValueChange={(v: any) => setCurrency(v)}>
   555→                <SelectTrigger><SelectValue /></SelectTrigger>
   556→                <SelectContent>
   557→                  {CURRENCIES.map(c => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}
   558→                </SelectContent>
   559→              </Select>
   560→            </div>
   561→          </div>
   562→
   563→          <div className="space-y-1.5">
   564→            <Label>{t('balance')}</Label>
   565→            <Input type="number" inputMode="decimal" value={balance} onChange={e => setBalance(e.target.value)} placeholder="0.00" />
   566→          </div>
   567→
   568→          <div className="space-y-1.5">
   569→            <Label>{lang === 'ar' ? 'اللون' : 'Color'}</Label>
   570→            <div className="flex gap-2 flex-wrap">

   572→                <button
   573→                  key={c}
   574→                  onClick={() => setColor(c)}
   575→                  className={cn(

   577→                    color === c ? 'border-slate-900 dark:border-white scale-110' : 'border-transparent'
